23    pub async fn look_up_username_hash(
24        &mut self,
25        hash: &[u8],
26    ) -> Result<Option<Aci>, ServiceError> {
27        use lookup_username_hash_response::Response;
28        let response =
29            AccountsAnonymousClient::new(self.grpc.unauthenticated())
30                .lookup_username_hash(LookupUsernameHashRequest {
31                    username_hash: hash.to_vec(),
32                })
33                .await?
34                .into_inner();
35        match response.response {
36            Some(Response::ServiceIdentifier(id)) => {
37                let uuid = uuid::Uuid::from_slice(&id.uuid).map_err(|_| {
38                    ServiceError::InvalidAddressType(ServiceIdKind::Aci)
39                })?;
40                Ok(Some(Aci::from(uuid)))
41            },
42            Some(Response::NotFound(_)) => Ok(None),
43            None => Err(ServiceError::InvalidFrame {
44                reason: "empty LookupUsernameHash response",
45            }),
46        }
47    }

49    /// Looks up the encrypted username stored at a username link handle and
50    /// decrypts it.
51    ///
52    /// `link` must be a full `https://signal.me/#eu/<payload>` URL. The payload
53    /// is the URL-safe base64 encoding of the 32-byte link entropy followed by
54    /// the 16-byte link handle UUID.
55    // Based on libsignal-net
56    pub async fn look_up_username_link(
57        &mut self,
58        link: &url::Url,
59    ) -> Result<Option<usernames::Username>, ServiceError> {
60        use lookup_username_link_response::Response;
61
62        let (uuid, entropy) = parse_username_link(link)?;
63
64        let response =
65            AccountsAnonymousClient::new(self.grpc.unauthenticated())
66                .lookup_username_link(LookupUsernameLinkRequest {
67                    username_link_handle: uuid.into_bytes().to_vec(),
68                })
69                .await?
70                .into_inner();
71        let encrypted_username = match response.response {
72            Some(Response::UsernameCiphertext(ciphertext)) => ciphertext,
73            Some(Response::NotFound(_)) => {
74                tracing::debug!("username link not found");
75                return Ok(None);
76            },
77            None => {
78                return Err(ServiceError::InvalidFrame {
79                    reason: "empty LookupUsernameLink response",
80                })
81            },
82        };
83
84        let plaintext_username =
85            usernames::decrypt_username(&entropy, &encrypted_username)
86                .map_err(|error| {
87                    tracing::error!(%error, "undecryptable username");
88                    ServiceError::InvalidFrame {
89                        reason: "undecryptable username link",
90                    }
91                })?;
92
93        let validated_username = usernames::Username::new(&plaintext_username).map_err(|e| {
94            // Exhaustively match UsernameError to make sure there's nothing we shouldn't log.
95            #[allow(clippy::let_unit_value)]
96            let _username_error_carries_no_information_that_would_be_bad_to_log = match e {
97                usernames::UsernameError::MissingSeparator
98                | usernames::UsernameError::NicknameCannotBeEmpty
99                | usernames::UsernameError::NicknameCannotStartWithDigit
100                | usernames::UsernameError::BadNicknameCharacter
101                | usernames::UsernameError::NicknameTooShort
102                | usernames::UsernameError::NicknameTooLong
103                | usernames::UsernameError::DiscriminatorCannotBeEmpty
104                | usernames::UsernameError::DiscriminatorCannotBeZero
105                | usernames::UsernameError::DiscriminatorCannotBeSingleDigit
106                | usernames::UsernameError::DiscriminatorCannotHaveLeadingZeros
107                | usernames::UsernameError::BadDiscriminatorCharacter
108                | usernames::UsernameError::DiscriminatorTooLarge => {}
109            };
110            tracing::warn!(error=%e, "username link decrypted to an invalid username");
111            tracing::debug!(error=%e,
112                "username link decrypted to '{plaintext_username}', which is not valid"
113            );
114            // The user didn't ever type this username, so the precise way in which it's invalid
115            // isn't important. Treat this equivalent to having found garbage data in the link. This
116            // simplifies error handling for callers.
117            ServiceError::InvalidFrame {
118                reason: "undecryptable username link",
119            }
120        })?;
121
122        Ok(Some(validated_username))
123    }
124}

126/// Splits a username link into its link handle UUID and link entropy.
127///
128/// `link` must be a full `https://signal.me/#eu/<payload>` URL. The payload is
129/// URL-safe base64 (no padding) of the 32-byte entropy followed by the 16-byte
130/// handle UUID.
131fn parse_username_link(
132    link: &url::Url,
133) -> Result<
134    (
135        uuid::Uuid,
136        [u8; usernames::constants::USERNAME_LINK_ENTROPY_SIZE],
137    ),
138    ServiceError,
139> {
140    if link.scheme() != "https" || link.host_str() != Some("signal.me") {
141        return Err(ServiceError::InvalidFrame {
142            reason: "username link base is not https://signal.me",
143        });
144    }
145
146    let fragment =
147        link.fragment().ok_or_else(|| ServiceError::InvalidFrame {
148            reason: "username link missing fragment",
149        })?;
150    let mut segments = fragment.split('/');
151    if segments.next() != Some("eu") {
152        return Err(ServiceError::InvalidFrame {
153            reason: "username link must start with #eu/",
154        });
155    }
156    let payload =
157        segments.next().ok_or_else(|| ServiceError::InvalidFrame {
158            reason: "username link payload missing",
159        })?;
160    if segments.next().is_some() {
161        return Err(ServiceError::InvalidFrame {
162            reason: "username link has extra path segments",
163        });
164    }
165
166    let bytes = BASE64_URL_SAFE_NO_PAD.decode(payload)?;
167
168    let (entropy, rest) = bytes
169        .split_first_chunk::<{ usernames::constants::USERNAME_LINK_ENTROPY_SIZE }>()
170        .ok_or_else(|| ServiceError::InvalidFrame {
171            reason: "username link payload shorter than entropy",
172        })?;
173
174    let handle_uuid = uuid::Uuid::from_slice(rest).map_err(|_| {
175        ServiceError::InvalidFrame {
176            reason: "username link payload missing handle UUID",
177        }
178    })?;
179
180    Ok((handle_uuid, *entropy))
181}

183impl SignalWebSocket<Identified> {
184    /// Sets the account's username link to encrypt `username` and returns the
185    /// shareable `https://signal.me/#eu/...` link.
186    ///
187    /// Generates fresh entropy and, unless `keep_link_handle` is `true` and the
188    /// account already has a link handle, a fresh server-assigned handle.
189    // Based on libsignal-net
190    pub async fn set_username_link(
191        &mut self,
192        username: &usernames::Username,
193        keep_link_handle: bool,
194    ) -> Result<url::Url, ServiceError> {
195        let (entropy, ciphertext) = usernames::create_for_username(
196            &mut rand::rng(),
197            username.to_string(),
198            None,
199        )
200        .map_err(|_e| ServiceError::InvalidFrame {
201            reason: "username too long to encrypt",
202        })?;
203
204        use set_username_link_response::Response;
205        let response = AccountsClient::new(self.grpc.authenticated()?)
206            .set_username_link(SetUsernameLinkRequest {
207                username_ciphertext: ciphertext,
208                keep_link_handle,
209            })
210            .await?
211            .into_inner();
212        let handle = match response.response {
213            Some(Response::UsernameLinkHandle(handle)) => {
214                uuid::Uuid::from_slice(&handle).map_err(|_| {
215                    ServiceError::InvalidFrame {
216                        reason: "invalid username link handle",
217                    }
218                })?
219            },
220            Some(Response::NoUsernameSet(_)) => {
221                return Err(ServiceError::UsernameHashNotSet)
222            },
223            None => {
224                return Err(ServiceError::InvalidFrame {
225                    reason: "empty SetUsernameLink response",
226                })
227            },
228        };
229
230        Ok(generate_username_link(handle, &entropy))
231    }
232}

234/// Builds the `https://signal.me/#eu/<base64url>` link from its parts.
235///
236/// `entropy` is the 32-byte link entropy; `handle` is the server-assigned
237/// link handle UUID. The payload is the URL-safe base64 (no padding) of
238/// `entropy || handle`.
239pub fn generate_username_link(
240    handle: uuid::Uuid,
241    entropy: &[u8; usernames::constants::USERNAME_LINK_ENTROPY_SIZE],
242) -> url::Url {
243    let mut payload = entropy.to_vec();
244    payload.extend_from_slice(handle.as_bytes());
245    let mut result = String::from("https://signal.me/#eu/");
246    BASE64_URL_SAFE_NO_PAD.encode_string(&payload, &mut result);
247
248    url::Url::parse(&result).expect("can only generate valid URLs")
249}

255    #[test]
256    fn generate_and_parse_link_round_trip() {
257        let entropy = [0x42; usernames::constants::USERNAME_LINK_ENTROPY_SIZE];
258        let handle = uuid::uuid!("9d0652a3-dcc3-4d11-975f-74d61598733f");
259        // deliberately chosen to round-trip cleanly through URL-safe base64
260        let link = generate_username_link(handle, &entropy);
261        assert!(link.as_str().starts_with("https://signal.me/#eu/"));
262
263        let (parsed_handle, parsed_entropy) =
264            parse_username_link(&link).unwrap();
265        assert_eq!(parsed_handle, handle);
266        assert_eq!(parsed_entropy, entropy);
267    }

275            .replace("signal.me", "example.com");
