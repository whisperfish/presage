103    pub async fn check_pre_keys<P: PreKeysStore>(
104        &mut self,
105        protocol_store: &mut P,
106        service_id_kind: ServiceIdKind,
107    ) -> Result<bool, ServiceError> {
108        // The server hashes the keys it holds, so these are the ids we last
109        // published, not the newest ones in store.
110        let Some(signed_prekey_id) =
111            protocol_store.active_signed_prekey_id().await?
112        else {
113            tracing::warn!("No signed prekey found");
114            return Ok(false);
115        };
116        let Some(kyber_prekey_id) =
117            protocol_store.active_last_resort_kyber_prekey_id().await?
118        else {
119            tracing::warn!("No last resort kyber prekey found");
120            return Ok(false);
121        };
122
123        // A missing active key is a mismatch to rotate away, not a failure.
124        let signed_prekey =
125            match protocol_store.get_signed_pre_key(signed_prekey_id).await {
126                Err(SignalProtocolError::InvalidSignedPreKeyId) => {
127                    tracing::warn!(
128                        ?signed_prekey_id,
129                        "Active signed prekey missing"
130                    );
131                    return Ok(false);
132                },
133                r => r?,
134            };
135        let kyber_prekey =
136            match protocol_store.get_kyber_pre_key(kyber_prekey_id).await {
137                Err(SignalProtocolError::InvalidKyberPreKeyId) => {
138                    tracing::warn!(
139                        ?kyber_prekey_id,
140                        "Active last resort kyber prekey missing"
141                    );
142                    return Ok(false);
143                },
144                r => r?,
145            };
146
147        // `SHA256(identityKeyBytes || signedEcPreKeyId || signedEcPreKeyIdBytes || lastResortKeyId || lastResortKeyBytes)`
148        let mut hash = Sha256::default();
149        hash.update(
150            protocol_store
151                .get_identity_key_pair()
152                .await?
153                .public_key()
154                .serialize(),
155        );
156        hash.update((u32::from(signed_prekey_id) as u64).to_be_bytes());
157        hash.update(signed_prekey.public_key()?.serialize());
158        hash.update((u32::from(kyber_prekey_id) as u64).to_be_bytes());
159        hash.update(kyber_prekey.public_key()?.serialize());
160
161        self.websocket
162            .check_pre_keys(service_id_kind, hash.finalize().as_ref())
163            .await
164    }

166    /// Bring one identity's pre-keys up to date.
167    ///
168    /// Two independent decisions:
169    ///
170    /// - The signed pre-key and last-resort Kyber pre-key are rotated on a
171    ///   local schedule (see
172    ///   [`signed_pre_keys_due_for_rotation`](PreKeysStore::signed_pre_keys_due_for_rotation)).
173    ///   Two keys, cheap enough to rotate regularly.
174    /// - One-time pre-keys are replenished only when the *server* reports it is
175    ///   running low, so a routine refresh usually uploads nothing here.
176    ///
177    /// Because the gates are separate, a fresh signed key never suppresses a
178    /// needed one-time replenish, and an empty server pool never forces an
179    /// early signed rotation.
180    ///
181    /// **Note**: callers should exhaust the message queue before calling this.
182    /// Replacing one-time keys races with messages that reference them —
183    /// though in practice the stale grace period and minimum-count floor in
184    /// [`clean_stale_pre_keys`](PreKeysStore::clean_stale_pre_keys) mitigate it.
185    pub async fn update_pre_key_bundle<P: PreKeysStore>(
186        &mut self,
187        protocol_store: &mut P,
188        service_id_kind: ServiceIdKind,
189    ) -> Result<(), ServiceError> {
190        if protocol_store.signed_pre_keys_due_for_rotation().await? {
191            self.rotate_signed_pre_keys(protocol_store, service_id_kind)
192                .await?;
193        } else {
194            tracing::trace!(?service_id_kind, "signed pre keys still current");
195        }
196
197        self.replenish_one_time_pre_keys(protocol_store, service_id_kind)
198            .await?;
199
200        protocol_store.clean_stale_pre_keys().await?;
201
202        Ok(())
203    }

205    /// Rotate the signed pre-key and last-resort kyber pre-key.
206    ///
207    /// Uploads two keys and leaves the server's one-time pool alone. Cheap
208    /// enough to run on a fixed schedule (upstream rotates roughly every two
209    /// days); the caller decides when, and a successful upload stamps the
210    /// rotation.
211    pub async fn rotate_signed_pre_keys<P: PreKeysStore>(
212        &mut self,
213        protocol_store: &mut P,
214        service_id_kind: ServiceIdKind,
215    ) -> Result<(), ServiceError> {
216        let identity_key_pair = protocol_store.get_identity_key_pair().await?;
217
218        let (signed_pre_key, pq_last_resort_key) =
219            crate::pre_keys::generate_signed_pre_keys(
220                protocol_store,
221                &mut rand::rng(),
222                &identity_key_pair,
223            )
224            .await?;
225
226        protocol_store
227            .store_signed_pre_key_bundle(&signed_pre_key, &pq_last_resort_key)
228            .await?;
229
230        tracing::info!(?service_id_kind, "rotating signed pre-key");
231
232        self.websocket
233            .register_pre_keys(
234                service_id_kind,
235                &PreKeyState {
236                    pre_keys: None,
237                    signed_pre_key: Some(SignedPreKeyEntity::try_from(
238                        &signed_pre_key,
239                    )?),
240                    identity_key: *identity_key_pair.identity_key(),
241                    pq_pre_keys: None,
242                    pq_last_resort_key: Some(KyberPreKeyEntity::try_from(
243                        &pq_last_resort_key,
244                    ))
245                    .transpose()?,
246                },
247            )
248            .await?;
249
250        protocol_store
251            .mark_signed_pre_keys_active(&signed_pre_key, &pq_last_resort_key)
252            .await?;
253
254        Ok(())
255    }

257    /// Replenish one-time pre-keys if the server is running low.
258    ///
259    /// Queries the server's remaining count and uploads a fresh batch only
260    /// when it has fallen below [`PRE_KEY_MINIMUM`]. Uploading replaces the
261    /// server's set, so the full batch is sent rather than a deficit.
262    ///
263    /// Returns whether an upload happened.
264    pub async fn replenish_one_time_pre_keys<P: PreKeysStore>(
265        &mut self,
266        protocol_store: &mut P,
267        service_id_kind: ServiceIdKind,
268    ) -> Result<bool, ServiceError> {
269        let status = self.websocket.get_pre_key_status(service_id_kind).await?;
270
271        let ec_needed = status.count < PRE_KEY_MINIMUM;
272        let kyber_needed = status.pq_count < PRE_KEY_MINIMUM;
273
274        if !ec_needed && !kyber_needed {
275            tracing::debug!(
276                ?service_id_kind,
277                count = status.count,
278                pq_count = status.pq_count,
279                "one-time pre-keys sufficient on server"
280            );
281            return Ok(false);
282        }
283
284        let identity_key_pair = protocol_store.get_identity_key_pair().await?;
285
286        let (pre_keys, pq_pre_keys) =
287            crate::pre_keys::generate_one_time_pre_keys(
288                protocol_store,
289                &mut rand::rng(),
290                &identity_key_pair,
291                if ec_needed { PRE_KEY_BATCH_SIZE } else { 0 },
292                if kyber_needed { PRE_KEY_BATCH_SIZE } else { 0 },
293            )
294            .await?;
295
296        protocol_store
297            .store_one_time_pre_key_bundle(&pre_keys, &pq_pre_keys)
298            .await?;
299
300        tracing::info!(
301            ?service_id_kind,
302            count = status.count,
303            pq_count = status.pq_count,
304            "replenishing one-time pre-keys"
305        );
306
307        self.websocket
308            .register_pre_keys(
309                service_id_kind,
310                &PreKeyState {
311                    pre_keys: ec_needed
312                        .then(|| {
313                            pre_keys
314                                .iter()
315                                .map(PreKeyEntity::try_from)
316                                .collect::<Result<_, _>>()
317                        })
318                        .transpose()?,
319                    signed_pre_key: None,
320                    identity_key: *identity_key_pair.identity_key(),
321                    pq_pre_keys: kyber_needed
322                        .then(|| {
323                            pq_pre_keys
324                                .iter()
325                                .map(KyberPreKeyEntity::try_from)
326                                .collect::<Result<_, _>>()
327                        })
328                        .transpose()?,
329                    pq_last_resort_key: None,
330                },
331            )
332            .await?;
333
334        Ok(true)
335    }

392    /// Link a new device, given a tsurl.
393    ///
394    /// Equivalent of Java's `AccountManager::addDevice()`
395    ///
396    /// When calling this, make sure that UnidentifiedDelivery is disabled, ie., that your
397    /// application does not send any unidentified messages before linking is complete.
398    /// Cfr.:
399    /// - `app/src/main/java/org/thoughtcrime/securesms/migrations/LegacyMigrationJob.java`:250 and;
400    /// - `app/src/main/java/org/thoughtcrime/securesms/DeviceActivity.java`:195
401    ///
402    /// ```java
403    /// TextSecurePreferences.setIsUnidentifiedDeliveryEnabled(context, false);
404    /// ```
405    pub async fn link_device<R: Rng + CryptoRng>(
406        &mut self,
407        csprng: &mut R,
408        url: url::Url,
409        aci_identity_store: &dyn IdentityKeyStore,
410        pni_identity_store: &dyn IdentityKeyStore,
411        secrets: ProvisioningSecrets,
412    ) -> Result<(), ProvisioningError> {
413        let ProvisioningSecrets {
414            credentials,
415            ephemeral_backup_key,
416            account_entropy_pool,
417            media_root_backup_key,
418            read_receipts,
419        } = secrets;
420
421        let query: HashMap<_, _> = url.query_pairs().collect();
422        let ephemeral_id =
423            query.get("uuid").ok_or(ProvisioningError::MissingUuid)?;
424        let pub_key = query
425            .get("pub_key")
426            .ok_or(ProvisioningError::MissingPublicKey)?;
427
428        let pub_key = BASE64_RELAXED
429            .decode(&**pub_key)
430            .map_err(|e| ProvisioningError::InvalidPublicKey(e.into()))?;
431        let pub_key = PublicKey::deserialize(&pub_key)
432            .map_err(|e| ProvisioningError::InvalidPublicKey(e.into()))?;
433
434        let aci_identity_key_pair =
435            aci_identity_store.get_identity_key_pair().await?;
436        let pni_identity_key_pair =
437            pni_identity_store.get_identity_key_pair().await?;
438
439        if credentials.aci.is_none() {
440            tracing::warn!("No local ACI set");
441        }
442        if credentials.pni.is_none() {
443            tracing::warn!("No local PNI set");
444        }
445
446        let provisioning_code = self.new_device_provisioning_code().await?;
447
448        // Same order as in Provisioning.proto for helping comparison.
449        let msg = ProvisionMessage {
450            aci_identity_key_public: Some(
451                aci_identity_key_pair.public_key().serialize().into_vec(),
452            ),
453            aci_identity_key_private: Some(
454                aci_identity_key_pair.private_key().serialize(),
455            ),
456            pni_identity_key_public: Some(
457                pni_identity_key_pair.public_key().serialize().into_vec(),
458            ),
459            pni_identity_key_private: Some(
460                pni_identity_key_pair.private_key().serialize(),
461            ),
462            aci: credentials.aci.as_ref().map(|u| u.to_string()),
463            pni: credentials.pni.as_ref().map(uuid::Uuid::to_string),
464            number: Some(credentials.e164()),
465            provisioning_code: Some(provisioning_code),
466            user_agent: None,
467            profile_key: self.profile_key.as_ref().map(|x| x.bytes.to_vec()),
468            read_receipts,
469            // CURRENT is not exposed by prost :(
470            provisioning_version: Some(i32::from(
471                ProvisioningVersion::TabletSupport,
472            ) as _),
473            ephemeral_backup_key: ephemeral_backup_key.map(|k| k.to_vec()), // 32-bytes
474            account_entropy_pool: Some(account_entropy_pool.to_string()),
475            media_root_backup_key: media_root_backup_key.map(|k| k.to_vec()), // 32-bytes
476            aci_binary: credentials.aci.map(|u| u.into_bytes().into()),
477            pni_binary: credentials.pni.map(|u| u.into_bytes().into()),
478        };
479
480        let cipher = ProvisioningCipher::from_public(pub_key);
481
482        let encrypted = cipher.encrypt(csprng, msg)?;
483        self.send_provisioning_message(ephemeral_id, encrypted)
484            .await?;
485        Ok(())
486    }

580    /// Upload a profile
581    ///
582    /// Panics if no `profile_key` was set.
583    ///
584    /// Returns the avatar url path.
585    pub async fn upload_versioned_profile<
586        's,
587        C: std::io::Read + Send + 's,
588        R: Rng + CryptoRng,
589        S: AsRef<str>,
590    >(
591        &mut self,
592        aci: libsignal_protocol::Aci,
593        name: ProfileName<S>,
594        about: Option<String>,
595        about_emoji: Option<String>,
596        avatar: AvatarWrite<&'s mut C>,
597        csprng: &mut R,
598    ) -> Result<Option<String>, ProfileManagerError> {
599        let profile_key =
600            self.profile_key.expect("set profile key in AccountManager");
601        let profile_cipher = ProfileCipher::new(profile_key);
602
603        // Profile encryption
604        let name = profile_cipher.encrypt_name(name.as_ref(), csprng)?;
605        let about = about.unwrap_or_default();
606        let about = profile_cipher.encrypt_about(about, csprng)?;
607        let about_emoji = about_emoji.unwrap_or_default();
608        let about_emoji = profile_cipher.encrypt_emoji(about_emoji, csprng)?;
609
610        // If avatar -> upload
611        if matches!(avatar, AvatarWrite::NewAvatar(_)) {
612            // FIXME ProfileCipherOutputStream.java
613            // It's just AES GCM, but a bit of work to decently implement it with a stream.
614            unimplemented!("Setting avatar requires ProfileCipherStream")
615        }
616
617        let profile_key = profile_cipher.into_inner();
618        let commitment = profile_key.get_commitment(aci);
619        let profile_key_version = profile_key.get_profile_key_version(aci);
620
621        Ok(self
622            .websocket
623            .write_profile::<C, S>(
624                &profile_key_version,
625                &name,
626                &about,
627                &about_emoji,
628                &commitment,
629                avatar,
630            )
631            .await?)
632    }

654    /// Update (encrypted) device name
655    pub async fn update_device_name<R: Rng + CryptoRng>(
656        &mut self,
657        device_id: libsignal_core::DeviceId,
658        device_name: &str,
659        aci: Aci,
660        aci_identity_store: &dyn IdentityKeyStore,
661        csprng: &mut R,
662    ) -> Result<(), ServiceError> {
663        let addr = aci.to_protocol_address(device_id).unwrap();
664        let public_key = aci_identity_store.get_identity(&addr).await?;
665        let Some(public_key) = public_key else {
666            return Err(ServiceError::SendError {
667                reason: format!("public key for device {addr:?} not found"),
668            });
669        };
670        let encrypted_device_name =
671            encrypt_device_name(csprng, device_name, &public_key)?;
672
673        #[derive(Serialize)]
674        #[serde(rename_all = "camelCase")]

731fn calculate_hmac256(
732    mac_key: &[u8],
733    ciphertext: &[u8],
734) -> Result<Output<Hmac<Sha256>>, ServiceError> {
735    let mut mac = Hmac::<Sha256>::new_from_slice(mac_key)
736        .map_err(|_| ServiceError::MacError)?;
737    mac.update(ciphertext);
738    Ok(mac.finalize().into_bytes())
739}

741pub fn encrypt_device_name<R: rand::Rng + rand::CryptoRng>(
742    csprng: &mut R,
743    device_name: &str,
744    identity_public: &IdentityKey,
745) -> Result<DeviceName, ServiceError> {
746    let plaintext = device_name.as_bytes().to_vec();
747    let ephemeral_key_pair = KeyPair::generate(csprng);
748
749    let master_secret = ephemeral_key_pair
750        .private_key
751        .calculate_agreement(identity_public.public_key())?;
752
753    let key1 = calculate_hmac256(&master_secret, b"auth")?;
754    let synthetic_iv = calculate_hmac256(&key1, &plaintext)?;
755    let synthetic_iv = &synthetic_iv[..16];
756
757    let key2 = calculate_hmac256(&master_secret, b"cipher")?;
758    let cipher_key = calculate_hmac256(&key2, synthetic_iv)?;
759
760    let mut ciphertext = plaintext;
761
762    const IV: [u8; 16] = [0; 16];
763    let mut cipher = Aes256Ctr128BE::new(&cipher_key, &IV.into());
764    cipher.apply_keystream(&mut ciphertext);
765
766    let device_name = DeviceName {
767        ephemeral_public: Some(
768            ephemeral_key_pair.public_key.serialize().to_vec(),
769        ),
770        synthetic_iv: Some(synthetic_iv.to_vec()),
771        ciphertext: Some(ciphertext),
772    };
773
774    Ok(device_name)
775}

777fn decrypt_device_name_from_device_info(
778    string: &str,
779    aci: &IdentityKeyPair,
780) -> Result<String, ServiceError> {
781    let Ok(data) = BASE64_RELAXED.decode(string) else {
782        tracing::trace!(
783            "Device name not base64 encoded, assuming plaintext: {}",
784            string.to_string()
785        );
786        return Ok(string.to_string());
787    };
788    let name = match DeviceName::decode(data.as_slice()) {
789        Ok(decoded) => decoded,
790        Err(_) => {
791            tracing::trace!(
792                "Decoding device name failed, assuming plaintext: {}",
793                string.to_string()
794            );
795            return Ok(string.to_string());
796        },
797    };
798    crate::decrypt_device_name(aci.private_key(), &name)
799}

801// Analogous to https://github.com/signalapp/Signal-Android/blob/d88a862e0985cc2bbc463c5f504f5bb4e91ad4fc/app/src/main/java/org/thoughtcrime/securesms/linkdevice/LinkDeviceRepository.kt#L121.
802fn decrypt_device_created_at_from_device_info(
803    id: DeviceId,
804    registration_id: i32,
805    string: &str,
806    aci: &IdentityKeyPair,
807) -> Result<chrono::DateTime<chrono::Utc>, ServiceError> {
808    use signal_crypto::SimpleHpkeReceiver;
809
810    let mut associated_data = [0; 5];
811    associated_data[0] = id.into();
812    associated_data[1..].copy_from_slice(&registration_id.to_be_bytes());
813
814    let data = BASE64_RELAXED.decode(string)?;
815
816    let result =
817        aci.private_key()
818            .open(b"deviceCreatedAt", &associated_data, &data)?;
819
820    let timestamp = i64::from_be_bytes(result.try_into().map_err(|_| {
821        ServiceError::DecryptDeviceInfoFieldError("created-at")
822    })?);
823
824    chrono::DateTime::<chrono::Utc>::from_timestamp_millis(timestamp)
825        .ok_or(ServiceError::DecryptDeviceInfoFieldError("created-at"))
826}

828pub fn decrypt_device_name(
829    private_key: &PrivateKey,
830    device_name: &DeviceName,
831) -> Result<String, ServiceError> {
832    let DeviceName {
833        ephemeral_public: Some(ephemeral_public),
834        synthetic_iv: Some(synthetic_iv),
835        ciphertext: Some(ciphertext),
836    } = device_name
837    else {
838        return Err(ServiceError::DecryptDeviceInfoFieldError("name"));
839    };
840
841    let synthetic_iv: [u8; 16] = synthetic_iv[..synthetic_iv.len().min(16)]
842        .try_into()
843        .map_err(|_| ServiceError::MacError)?;
844
845    let ephemeral_public = PublicKey::deserialize(ephemeral_public)?;
846
847    let master_secret = private_key.calculate_agreement(&ephemeral_public)?;
848    let key2 = calculate_hmac256(&master_secret, b"cipher")?;
849    let cipher_key = calculate_hmac256(&key2, &synthetic_iv)?;
850
851    let mut plaintext = ciphertext.to_vec();
852    const IV: [u8; 16] = [0; 16];
853    let mut cipher = Aes256Ctr128BE::new(&cipher_key, &IV.into());
854    cipher.apply_keystream(&mut plaintext);
855
856    let key1 = calculate_hmac256(&master_secret, b"auth")?;
857    let our_synthetic_iv = calculate_hmac256(&key1, &plaintext)?;
858    let our_synthetic_iv = &our_synthetic_iv[..16];
859
860    if synthetic_iv != our_synthetic_iv {
861        Err(ServiceError::MacError)
862    } else {
863        Ok(String::from_utf8_lossy(&plaintext).to_string())
864    }
865}
