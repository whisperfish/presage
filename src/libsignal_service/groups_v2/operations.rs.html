37#[derive(Debug, thiserror::Error)]
38pub enum GroupDecodingError {
39    #[error("zero-knowledge group deserialization failure")]
40    ZkGroupDeserializationFailure,
41    #[error("zero-knowledge group verification failure")]
42    ZkGroupVerificationFailure,
43    #[error(transparent)]
44    BincodeError(#[from] bincode::Error),
45    #[error("protobuf message decoding error: {0}")]
46    ProtobufDecodeError(#[from] prost::DecodeError),
47    #[error("wrong group attribute blob")]
48    WrongBlob,
49    #[error("wrong enum value")]
50    WrongEnumValue,
51    #[error("wrong service ID type: should be ACI")]
52    NotAci,
53    #[error("wrong service ID type: should be PNI")]
54    NotPni,
55}

71        &self,
72        service_id: ServiceId,

79    fn decrypt_service_id(
80        &self,
81        ciphertext: &[u8],
82    ) -> Result<ServiceId, GroupDecodingError> {
83        match self
84            .group_secret_params
85            .decrypt_service_id(zkgroup::deserialize(ciphertext)?)?
86        {
87            ServiceId::Aci(aci) => Ok(ServiceId::from(aci)),
88            ServiceId::Pni(pni) => Ok(ServiceId::from(pni)),

156    fn decrypt_profile_key_presentation(
157        &self,
158        aci: &[u8],
159        profile_key: &[u8],
160        presentation: &[u8],
161    ) -> Result<(Aci, ProfileKey), GroupDecodingError> {
162        if presentation.is_empty() {
163            let aci = self.decrypt_aci(aci)?;
164            let profile_key = self.decrypt_profile_key(profile_key, aci)?;
165            return Ok((aci, profile_key));
166        }
167
168        let profile_key_credential_presentation =
169            AnyProfileKeyCredentialPresentation::new(presentation)?;
170
171        match self.group_secret_params.decrypt_service_id(
172            profile_key_credential_presentation.get_uuid_ciphertext(),
173        )? {
174            ServiceId::Aci(aci) => {
175                let profile_key =
176                    self.group_secret_params.decrypt_profile_key(
177                        profile_key_credential_presentation
178                            .get_profile_key_ciphertext(),
179                        aci,
180                    )?;
181                Ok((aci, profile_key))
182            },
183            _ => Err(GroupDecodingError::NotAci),
184        }
185    }

276    /// Decrypts an optional string field, returning `None` when `bytes` is empty
277    /// so absent labels don't fail the whole decode.
278    fn maybe_decrypt_string(
279        &self,
280        bytes: &[u8],
281    ) -> Result<Option<String>, GroupDecodingError> {
282        if bytes.is_empty() {
283            return Ok(None);
284        }
285        self.decrypt_string(bytes).map(Some)
286    }

312    fn decrypt_blob(&self, bytes: &[u8]) -> GroupAttributeBlob {
313        if bytes.is_empty() {
314            GroupAttributeBlob::default()
315        } else if bytes.len() < 29 {
316            tracing::warn!("bad encrypted blob length");
317            GroupAttributeBlob::default()
318        } else {
319            self.group_secret_params
320                .decrypt_blob_with_padding(bytes)
321                .map_err(GroupDecodingError::from)
322                .and_then(|plaintext| {
323                    GroupAttributeBlob::decode(Bytes::from(plaintext))
324                        .map_err(GroupDecodingError::ProtobufDecodeError)
325                })
326                .unwrap_or_else(|e| {
327                    tracing::warn!("bad encrypted blob: {}", e);
328                    GroupAttributeBlob::default()
329                })
330        }
331    }

333    /// Helper method to encrypt a `group_attribute_blob::Content`.
334    ///
335    /// # Padding Format
336    ///
337    /// Uses `encrypt_blob_with_padding` format from Signal's zkgroup's
338    /// `GroupSecretParams`, which prepends a 4-byte big-endian padding length value
339    /// to the plaintext before encryption. For group attribute blobs, padding is
340    /// always 0, so the format is:
341    /// - First 4 bytes: `0u32.to_be_bytes()` (padding length = 0)
342    /// - Remaining bytes: protobuf-encoded `GroupAttributeBlob`
343    ///
344    /// # References
345    ///
346    /// - Signal libsignal repository: <https://github.com/signalapp/libsignal>
347    /// - GroupSecretParams implementation:
348    ///   `rust/zkgroup/src/api/groups/group_params.rs`
349    /// - Java ClientZkGroupCipher usage:
350    ///   `java/shared/java/org/signal/libsignal/zkgroup/groups/ClientZkGroupCipher.java`
351    fn encrypt_blob_content<R: rand::Rng + rand::CryptoRng>(
352        &self,
353        content: group_attribute_blob::Content,
354        rng: &mut R,
355    ) -> Vec<u8> {
356        let blob = GroupAttributeBlob {
357            content: Some(content),
358        };
359        let buf = blob.encode_to_vec();
360
361        let mut randomness = [0u8; 32];
362        rng.fill_bytes(&mut randomness);
363        self.group_secret_params
364            .encrypt_blob_with_padding(randomness, &buf, 0)
365    }

406    fn decrypt_title(&self, ciphertext: &[u8]) -> String {
407        use group_attribute_blob::Content;
408        match self.decrypt_blob(ciphertext).content {
409            Some(Content::Title(title)) => title,
410            _ => "".into(),
411        }
412    }
413
414    fn decrypt_description_text(&self, ciphertext: &[u8]) -> Option<String> {
415        use group_attribute_blob::Content;
416        match self.decrypt_blob(ciphertext).content {
417            Some(Content::DescriptionText(d)) => {
418                Some(d).filter(|d| !d.is_empty())
419            },
420            _ => None,
421        }
422    }
423
424    fn decrypt_disappearing_messages_timer(
425        &self,
426        ciphertext: &[u8],
427    ) -> Option<Timer> {
428        use group_attribute_blob::Content;
429        match self.decrypt_blob(ciphertext).content {
430            Some(Content::DisappearingMessagesDuration(duration)) => {
431                Some(Timer { duration })
432            },
433            _ => None,
434        }
435    }

443    pub fn decrypt_group(
444        &self,
445        group: proto::Group,
446    ) -> Result<Group, GroupDecodingError> {
447        // Destructuring to catch any future changes
448        let proto::Group {
449            public_key: _,
450            title,
451            avatar_url,
452            disappearing_messages_timer,
453            access_control,
454            version,
455            members,
456            members_pending_profile_key,
457            members_pending_admin_approval,
458            invite_link_password,
459            description,
460            announcements_only,
461            members_banned,
462            terminated,
463        } = group;
464
465        let title = self.decrypt_title(&title);
466
467        let description_text = self.decrypt_description_text(&description);
468
469        let disappearing_messages_timer = self
470            .decrypt_disappearing_messages_timer(&disappearing_messages_timer);
471
472        let members = members
473            .into_iter()
474            .map(|m| self.decrypt_member(m))
475            .collect::<Result<_, _>>()?;
476
477        let members_pending_profile_key = members_pending_profile_key
478            .into_iter()
479            .map(|m| self.decrypt_pending_member(m))
480            .collect::<Result<_, _>>()?;
481
482        let members_pending_admin_approval = members_pending_admin_approval
483            .into_iter()
484            .map(|m| self.decrypt_requesting_member(m))
485            .collect::<Result<_, _>>()?;
486
487        let members_banned = members_banned
488            .into_iter()
489            .map(|m| self.decrypt_banned_member(m))
490            .collect::<Result<_, _>>()?;
491
492        let access_control =
493            access_control.map(TryInto::try_into).transpose()?;
494
495        Ok(Group {
496            title,
497            avatar: avatar_url,
498            disappearing_messages_timer,
499            access_control,
500            version,
501            members,
502            members_pending_profile_key,
503            members_pending_admin_approval,
504            invite_link_password,
505            description_text,
506            announcements_only,
507            members_banned,
508            terminated,
509        })
510    }

512    pub fn decrypt_group_change(
513        &self,
514        group_change: proto::GroupChange,
515    ) -> Result<GroupChanges, GroupDecodingError> {
516        // Destructuring to catch any future changes
517        let proto::GroupChange {
518            actions,
519            server_signature: _,
520            change_epoch,
521        } = group_change;
522
523        let proto::group_change::Actions {
524            group_id,
525            source_user_id,
526            version,
527            add_members,
528            delete_members,
529            modify_member_roles,
530            modify_member_profile_keys,
531            add_members_pending_profile_key,
532            delete_members_pending_profile_key,
533            promote_members_pending_profile_key,
534            modify_title,
535            modify_avatar,
536            modify_disappearing_message_timer,
537            modify_attributes_access,
538            modify_member_access,
539            modify_add_from_invite_link_access,
540            add_members_pending_admin_approval,
541            delete_members_pending_admin_approval,
542            promote_members_pending_admin_approval,
543            modify_invite_link_password,
544            modify_description,
545            modify_announcements_only,
546            add_members_banned,
547            delete_members_banned,
548            promote_members_pending_pni_aci_profile_key,
549            modify_member_labels,
550            modify_member_label_access,
551            terminate_group,
552        } = Message::decode(Bytes::from(actions))?;
553
554        let source_user_id = self.decrypt_aci(&source_user_id)?;
555
556        let new_members =
557            add_members
558                .into_iter()
559                .filter_map(|m| m.added)
560                .map(|added| {
561                    Ok(GroupChange::NewMember(self.decrypt_member(added)?))
562                });
563
564        let delete_members = delete_members.into_iter().map(|c| {
565            Ok(GroupChange::DeleteMember(
566                self.decrypt_aci(&c.deleted_user_id)?,
567            ))
568        });
569
570        let modify_member_roles = modify_member_roles.into_iter().map(|m| {
571            Ok(GroupChange::ModifyMemberRole {
572                aci: self.decrypt_aci(&m.user_id)?,
573                role: m.role.try_into()?,
574            })
575        });
576
577        let modify_member_profile_keys =
578            modify_member_profile_keys.into_iter().map(|m| {
579                let (aci, profile_key) = self
580                    .decrypt_profile_key_presentation(
581                        &m.user_id,
582                        &m.profile_key,
583                        &m.presentation,
584                    )?;
585                Ok(GroupChange::ModifyMemberProfileKey { aci, profile_key })
586            });
587
588        let add_members_pending_profile_key = add_members_pending_profile_key
589            .into_iter()
590            .filter_map(|m| m.added)
591            .map(|added| {
592                Ok(GroupChange::NewPendingMember(
593                    self.decrypt_pending_member(added)?,
594                ))
595            });
596
597        let delete_members_pending_profile_key =
598            delete_members_pending_profile_key.into_iter().map(|m| {
599                Ok(GroupChange::DeletePendingMember(
600                    self.decrypt_service_id(&m.deleted_user_id)?,
601                ))
602            });
603
604        let promote_members_pending_profile_key =
605            promote_members_pending_profile_key.into_iter().map(|m| {
606                let (aci, profile_key) = self
607                    .decrypt_profile_key_presentation(
608                        &m.user_id,
609                        &m.profile_key,
610                        &m.presentation,
611                    )?;
612                Ok(GroupChange::PromotePendingMember {
613                    address: aci.into(),
614                    profile_key,
615                })
616            });
617
618        let modify_title = modify_title
619            .into_iter()
620            .map(|m| Ok(GroupChange::Title(self.decrypt_title(&m.title))));
621
622        let modify_avatar = modify_avatar
623            .into_iter()
624            .map(|m| Ok(GroupChange::Avatar(m.avatar)));
625
626        let modify_description = modify_description.into_iter().map(|m| {
627            Ok(GroupChange::Description(
628                self.decrypt_description_text(&m.description),
629            ))
630        });
631
632        let modify_disappearing_message_timer =
633            modify_disappearing_message_timer.into_iter().map(|m| {
634                Ok(GroupChange::Timer(
635                    self.decrypt_disappearing_messages_timer(&m.timer),
636                ))
637            });
638
639        let modify_attributes_access =
640            modify_attributes_access.into_iter().map(|m| {
641                Ok(GroupChange::AttributeAccess(
642                    m.attributes_access.try_into()?,
643                ))
644            });
645
646        let modify_member_access = modify_member_access.into_iter().map(|m| {
647            Ok(GroupChange::MemberAccess(m.members_access.try_into()?))
648        });
649
650        let add_members_banned = add_members_banned
651            .into_iter()
652            .filter_map(|m| m.added)
653            .map(|m| {
654                Ok(GroupChange::AddBannedMember(self.decrypt_banned_member(m)?))
655            });
656
657        let delete_members_banned =
658            delete_members_banned.into_iter().map(|m| {
659                Ok(GroupChange::DeleteBannedMember(
660                    self.decrypt_service_id(&m.deleted_user_id)?,
661                ))
662            });
663
664        let promote_members_pending_pni_aci_profile_key =
665            promote_members_pending_pni_aci_profile_key
666                .into_iter()
667                .map(|m| {
668                    let promoted =
669                        self.decrypt_pni_aci_promotion_presentation(&m)?;
670                    Ok(GroupChange::PromotePendingPniAciMemberProfileKey(
671                        promoted,
672                    ))
673                });
674
675        let modify_add_from_invite_link_access =
676            modify_add_from_invite_link_access.into_iter().map(|m| {
677                Ok(GroupChange::InviteLinkAccess(
678                    m.add_from_invite_link_access.try_into()?,
679                ))
680            });
681
682        let add_members_pending_admin_approval =
683            add_members_pending_admin_approval
684                .into_iter()
685                .filter_map(|m| m.added)
686                .map(|added| {
687                    Ok(GroupChange::NewRequestingMember(
688                        self.decrypt_requesting_member(added)?,
689                    ))
690                });
691
692        let delete_members_pending_admin_approval =
693            delete_members_pending_admin_approval.into_iter().map(|m| {
694                Ok(GroupChange::DeleteRequestingMember(
695                    self.decrypt_aci(&m.deleted_user_id)?,
696                ))
697            });
698
699        let promote_members_pending_admin_approval =
700            promote_members_pending_admin_approval.into_iter().map(|m| {
701                Ok(GroupChange::PromoteRequestingMember {
702                    aci: self.decrypt_aci(&m.user_id)?,
703                    role: m.role.try_into()?,
704                })
705            });
706
707        let modify_invite_link_password =
708            modify_invite_link_password.into_iter().map(|m| {
709                Ok(GroupChange::InviteLinkPassword(
710                    BASE64_RELAXED.encode(m.invite_link_password),
711                ))
712            });
713
714        let modify_announcements_only = modify_announcements_only
715            .into_iter()
716            .map(|m| Ok(GroupChange::AnnouncementOnly(m.announcements_only)));
717
718        let modify_member_labels = modify_member_labels.into_iter().map(|m| {
719            Ok(GroupChange::MemberLabel {
720                user_id: self.decrypt_service_id(&m.user_id)?,
721                label_emoji: self.decrypt_member_label_emoji(&m.label_emoji),
722                label_string: self.decrypt_member_label_text(&m.label_string),
723            })
724        });
725
726        let modify_member_label_access =
727            modify_member_label_access.into_iter().map(|m| {
728                Ok(GroupChange::MemberLabelAccess(
729                    m.member_label_access.try_into()?,
730                ))
731            });
732
733        let terminate_group = terminate_group.into_iter().map(|_m| {
734            // The *presence* of the field signifies termination.
735            // TerminateGroupAction is a unit struct.
736            Ok(GroupChange::TerminateGroup)
737        });
738
739        let changes: Result<Vec<GroupChange>, GroupDecodingError> = new_members
740            .chain(delete_members)
741            .chain(modify_member_roles)
742            .chain(modify_member_profile_keys)
743            .chain(add_members_pending_profile_key)
744            .chain(delete_members_pending_profile_key)
745            .chain(promote_members_pending_profile_key)
746            .chain(modify_title)
747            .chain(modify_avatar)
748            .chain(modify_disappearing_message_timer)
749            .chain(modify_attributes_access)
750            .chain(modify_description)
751            .chain(modify_member_access)
752            .chain(add_members_banned)
753            .chain(delete_members_banned)
754            .chain(promote_members_pending_pni_aci_profile_key)
755            .chain(modify_add_from_invite_link_access)
756            .chain(add_members_pending_admin_approval)
757            .chain(delete_members_pending_admin_approval)
758            .chain(promote_members_pending_admin_approval)
759            .chain(modify_invite_link_password)
760            .chain(modify_announcements_only)
761            .chain(modify_member_labels)
762            .chain(modify_member_label_access)
763            .chain(terminate_group)
764            .collect();
765
766        Ok(GroupChanges {
767            group_id: group_id
768                .try_into()
769                .map_err(|_| GroupDecodingError::WrongBlob)?,
770            editor: source_user_id,
771            version,
772            changes: changes?,
773            change_epoch,
774        })
775    }

872    /// Encrypt a group for creation, using credentials for member presentations.
873    ///
874    /// This method properly populates the `presentation` field for members with
875    /// credentials, which is required by the Signal server for group creation.
876    ///
877    /// Members with credentials get added with presentations (full members).
878    /// Members without credentials get added as pending invites.
879    ///
880    /// # Arguments
881    /// * `title` - The group title
882    /// * `description` - Optional group description
883    /// * `disappearing_messages_timer` - Optional disappearing messages timer
884    /// * `access_control` - Optional access control settings
885    /// * `self_credential` - The creator's own credential (required)
886    /// * `avatar_url` - The group avatar URL
887    /// * `member_candidates` - Other members to add, with optional credentials
888    /// * `server_public_params` - Server public params for creating presentations
889    /// * `rng` - Random number generator
890    #[allow(clippy::too_many_arguments)]
891    pub fn encrypt_group_with_credentials<R: rand::Rng + rand::CryptoRng>(
892        &self,
893        title: &str,
894        description: Option<&str>,
895        disappearing_messages_timer: Option<&Timer>,
896        access_control: Option<&AccessControl>,
897        self_credential: &ExpiringProfileKeyCredential,
898        member_candidates: &[GroupMemberCandidate],
899        server_public_params: &ServerPublicParams,
900        avatar_url: String,
901        rng: &mut R,
902    ) -> Result<proto::Group, GroupDecodingError> {
903        let mut members = Vec::new();
904        let mut members_pending_profile_key = Vec::new();
905
906        // Add self as administrator with presentation
907        let self_presentation = self
908            .create_member_presentation(server_public_params, self_credential);
909        members.push(proto::Member {
910            user_id: vec![],     // Server extracts from presentation
911            profile_key: vec![], // Server extracts from presentation
912            presentation: self_presentation,
913            role: proto::member::Role::Administrator.into(),
914            joined_at_version: 0,
915            label_emoji: vec![],
916            label_string: vec![],
917        });
918
919        // Add other members
920        for candidate in member_candidates {
921            if let Some(credential) = &candidate.credential {
922                // Has credential - add as full member with presentation
923                let presentation = self.create_member_presentation(
924                    server_public_params,
925                    credential,
926                );
927                members.push(proto::Member {
928                    user_id: vec![],
929                    profile_key: vec![],
930                    presentation,
931                    role: proto::member::Role::Default.into(),
932                    joined_at_version: 0,
933                    label_emoji: vec![],
934                    label_string: vec![],
935                });
936            } else {
937                // No credential - add as pending invite
938                let user_id_ciphertext =
939                    self.encrypt_service_id(candidate.service_id)?;
940                let self_aci = self_credential.aci();
941                members_pending_profile_key.push(
942                    proto::MemberPendingProfileKey {
943                        member: Some(proto::Member {
944                            user_id: user_id_ciphertext,
945                            profile_key: vec![],
946                            presentation: vec![],
947                            role: proto::member::Role::Default.into(),
948                            joined_at_version: 0,
949                            label_emoji: vec![],
950                            label_string: vec![],
951                        }),
952                        added_by_user_id: self.encrypt_aci(self_aci)?,
953                        timestamp: 0, // Server sets
954                    },
955                );
956            }
957        }
958
959        // Encrypt title, description, timer
960        let encrypted_title = self.encrypt_title(title, rng);
961        let encrypted_description = self.encrypt_description(description, rng);
962        let encrypted_timer = self.encrypt_disappearing_messages_timer(
963            disappearing_messages_timer,
964            rng,
965        );
966
967        // Convert access control
968        let proto_access_control =
969            access_control.map(|ac| proto::AccessControl {
970                attributes: ac.attributes.into(),
971                members: ac.members.into(),
972                add_from_invite_link: ac.add_from_invite_link.into(),
973                member_label: ac.member_label.into(),
974            });
975
976        Ok(proto::Group {
977            public_key: zkgroup::serialize(
978                &self.group_secret_params.get_public_params(),
979            ),
980            title: encrypted_title,
981            avatar_url,
982            disappearing_messages_timer: encrypted_timer,
983            access_control: proto_access_control,
984            version: 0,
985            members,
986            members_pending_profile_key,
987            members_pending_admin_approval: vec![],
988            invite_link_password: vec![],
989            description: encrypted_description,
990            announcements_only: false,
991            members_banned: vec![],
992            terminated: false,
993        })
994    }
