38#[derive(Debug, thiserror::Error)]
39pub enum GroupDecodingError {
40    #[error("zero-knowledge group deserialization failure")]
41    ZkGroupDeserializationFailure,
42    #[error("zero-knowledge group verification failure")]
43    ZkGroupVerificationFailure,
44    #[error(transparent)]
45    BincodeError(#[from] bincode::Error),
46    #[error("protobuf message decoding error: {0}")]
47    ProtobufDecodeError(#[from] prost::DecodeError),
48    #[error("wrong group attribute blob")]
49    WrongBlob,
50    #[error("wrong enum value")]
51    WrongEnumValue,
52    #[error("wrong service ID type: should be ACI")]
53    NotAci,
54    #[error("wrong service ID type: should be PNI")]
55    NotPni,
56}

157    fn decrypt_profile_key_presentation(
158        &self,
159        aci: &[u8],
160        profile_key: &[u8],
161        presentation: &[u8],
162    ) -> Result<(Aci, ProfileKey), GroupDecodingError> {
163        if presentation.is_empty() {
164            let aci = self.decrypt_aci(aci)?;
165            let profile_key = self.decrypt_profile_key(profile_key, aci)?;
166            return Ok((aci, profile_key));
167        }
168
169        let profile_key_credential_presentation =
170            AnyProfileKeyCredentialPresentation::new(presentation)?;
171
172        match self.group_secret_params.decrypt_service_id(
173            profile_key_credential_presentation.get_uuid_ciphertext(),
174        )? {
175            ServiceId::Aci(aci) => {
176                let profile_key =
177                    self.group_secret_params.decrypt_profile_key(
178                        profile_key_credential_presentation
179                            .get_profile_key_ciphertext(),
180                        aci,
181                    )?;
182                Ok((aci, profile_key))
183            },
184            _ => Err(GroupDecodingError::NotAci),
185        }
186    }

277    /// Decrypts an optional string field, returning `None` when `bytes` is empty
278    /// so absent labels don't fail the whole decode.
279    fn maybe_decrypt_string(
280        &self,
281        bytes: &[u8],
282    ) -> Result<Option<String>, GroupDecodingError> {
283        if bytes.is_empty() {
284            return Ok(None);
285        }
286        self.decrypt_string(bytes).map(Some)
287    }

313    fn decrypt_blob(&self, bytes: &[u8]) -> GroupAttributeBlob {
314        if bytes.is_empty() {
315            GroupAttributeBlob::default()
316        } else if bytes.len() < 29 {
317            tracing::warn!("bad encrypted blob length");
318            GroupAttributeBlob::default()
319        } else {
320            self.group_secret_params
321                .decrypt_blob_with_padding(bytes)
322                .map_err(GroupDecodingError::from)
323                .and_then(|plaintext| {
324                    GroupAttributeBlob::decode(Bytes::from(plaintext))
325                        .map_err(GroupDecodingError::ProtobufDecodeError)
326                })
327                .unwrap_or_else(|e| {
328                    tracing::warn!("bad encrypted blob: {}", e);
329                    GroupAttributeBlob::default()
330                })
331        }
332    }

334    /// Helper method to encrypt a `group_attribute_blob::Content`.
335    ///
336    /// # Padding Format
337    ///
338    /// Uses `encrypt_blob_with_padding` format from Signal's zkgroup's
339    /// `GroupSecretParams`, which prepends a 4-byte big-endian padding length value
340    /// to the plaintext before encryption. For group attribute blobs, padding is
341    /// always 0, so the format is:
342    /// - First 4 bytes: `0u32.to_be_bytes()` (padding length = 0)
343    /// - Remaining bytes: protobuf-encoded `GroupAttributeBlob`
344    ///
345    /// # References
346    ///
347    /// - Signal libsignal repository: <https://github.com/signalapp/libsignal>
348    /// - GroupSecretParams implementation:
349    ///   `rust/zkgroup/src/api/groups/group_params.rs`
350    /// - Java ClientZkGroupCipher usage:
351    ///   `java/shared/java/org/signal/libsignal/zkgroup/groups/ClientZkGroupCipher.java`
352    fn encrypt_blob_content<R: rand::Rng + rand::CryptoRng>(
353        &self,
354        content: group_attribute_blob::Content,
355        rng: &mut R,
356    ) -> Vec<u8> {
357        let blob = GroupAttributeBlob {
358            content: Some(content),
359        };
360        let buf = blob.encode_to_vec();
361
362        let mut randomness = [0u8; 32];
363        rng.fill_bytes(&mut randomness);
364        self.group_secret_params
365            .encrypt_blob_with_padding(randomness, &buf, 0)
366    }

418    fn decrypt_title(&self, ciphertext: &[u8]) -> String {
419        use group_attribute_blob::Content;
420        match self.decrypt_blob(ciphertext).content {
421            Some(Content::Title(title)) => title,
422            _ => "".into(),
423        }
424    }

426    fn decrypt_description_text(&self, ciphertext: &[u8]) -> Option<String> {
427        use group_attribute_blob::Content;
428        match self.decrypt_blob(ciphertext).content {
429            Some(Content::DescriptionText(d)) => {
430                Some(d).filter(|d| !d.is_empty())
431            },
432            _ => None,
433        }
434    }

455    pub fn decrypt_group(
456        &self,
457        group: proto::Group,
458    ) -> Result<Group, GroupDecodingError> {
459        // Destructuring to catch any future changes
460        let proto::Group {
461            public_key: _,
462            title,
463            avatar_url,
464            disappearing_messages_timer,
465            access_control,
466            version,
467            members,
468            members_pending_profile_key,
469            members_pending_admin_approval,
470            invite_link_password,
471            description,
472            announcements_only,
473            members_banned,
474            terminated,
475        } = group;
476
477        let title = self.decrypt_title(&title);
478
479        let description_text = self.decrypt_description_text(&description);
480
481        let disappearing_messages_timer = self
482            .decrypt_disappearing_messages_timer(&disappearing_messages_timer);
483
484        let members = members
485            .into_iter()
486            .map(|m| self.decrypt_member(m))
487            .collect::<Result<_, _>>()?;
488
489        let members_pending_profile_key = members_pending_profile_key
490            .into_iter()
491            .map(|m| self.decrypt_pending_member(m))
492            .collect::<Result<_, _>>()?;
493
494        let members_pending_admin_approval = members_pending_admin_approval
495            .into_iter()
496            .map(|m| self.decrypt_requesting_member(m))
497            .collect::<Result<_, _>>()?;
498
499        let members_banned = members_banned
500            .into_iter()
501            .map(|m| self.decrypt_banned_member(m))
502            .collect::<Result<_, _>>()?;
503
504        let access_control =
505            access_control.map(TryInto::try_into).transpose()?;
506
507        Ok(Group {
508            title,
509            avatar: avatar_url,
510            disappearing_messages_timer,
511            access_control,
512            version,
513            members,
514            members_pending_profile_key,
515            members_pending_admin_approval,
516            invite_link_password,
517            description_text,
518            announcements_only,
519            members_banned,
520            terminated,
521        })
522    }

539    pub fn decrypt_group_change(
540        &self,
541        group_change: proto::GroupChange,
542    ) -> Result<GroupChanges, GroupDecodingError> {
543        // Destructuring to catch any future changes
544        let proto::GroupChange {
545            actions,
546            server_signature: _,
547            change_epoch,
548        } = group_change;
549
550        let proto::group_change::Actions {
551            group_id,
552            source_user_id,
553            version,
554            add_members,
555            delete_members,
556            modify_member_roles,
557            modify_member_profile_keys,
558            add_members_pending_profile_key,
559            delete_members_pending_profile_key,
560            promote_members_pending_profile_key,
561            modify_title,
562            modify_avatar,
563            modify_disappearing_message_timer,
564            modify_attributes_access,
565            modify_member_access,
566            modify_add_from_invite_link_access,
567            add_members_pending_admin_approval,
568            delete_members_pending_admin_approval,
569            promote_members_pending_admin_approval,
570            modify_invite_link_password,
571            modify_description,
572            modify_announcements_only,
573            add_members_banned,
574            delete_members_banned,
575            promote_members_pending_pni_aci_profile_key,
576            modify_member_labels,
577            modify_member_label_access,
578            terminate_group,
579        } = Message::decode(Bytes::from(actions))?;
580
581        let source_user_id = self.decrypt_aci(&source_user_id)?;
582
583        let new_members =
584            add_members
585                .into_iter()
586                .filter_map(|m| m.added)
587                .map(|added| {
588                    Ok(GroupChange::NewMember(self.decrypt_member(added)?))
589                });
590
591        let delete_members = delete_members.into_iter().map(|c| {
592            Ok(GroupChange::DeleteMember(
593                self.decrypt_aci(&c.deleted_user_id)?,
594            ))
595        });
596
597        let modify_member_roles = modify_member_roles.into_iter().map(|m| {
598            Ok(GroupChange::ModifyMemberRole {
599                aci: self.decrypt_aci(&m.user_id)?,
600                role: m.role.try_into()?,
601            })
602        });
603
604        let modify_member_profile_keys =
605            modify_member_profile_keys.into_iter().map(|m| {
606                let (aci, profile_key) = self
607                    .decrypt_profile_key_presentation(
608                        &m.user_id,
609                        &m.profile_key,
610                        &m.presentation,
611                    )?;
612                Ok(GroupChange::ModifyMemberProfileKey { aci, profile_key })
613            });
614
615        let add_members_pending_profile_key = add_members_pending_profile_key
616            .into_iter()
617            .filter_map(|m| m.added)
618            .map(|added| {
619                Ok(GroupChange::NewPendingMember(
620                    self.decrypt_pending_member(added)?,
621                ))
622            });
623
624        let delete_members_pending_profile_key =
625            delete_members_pending_profile_key.into_iter().map(|m| {
626                Ok(GroupChange::DeletePendingMember(
627                    self.decrypt_service_id(&m.deleted_user_id)?,
628                ))
629            });
630
631        let promote_members_pending_profile_key =
632            promote_members_pending_profile_key.into_iter().map(|m| {
633                let (aci, profile_key) = self
634                    .decrypt_profile_key_presentation(
635                        &m.user_id,
636                        &m.profile_key,
637                        &m.presentation,
638                    )?;
639                Ok(GroupChange::PromotePendingMember {
640                    address: aci.into(),
641                    profile_key,
642                })
643            });
644
645        let modify_title = modify_title
646            .into_iter()
647            .map(|m| Ok(GroupChange::Title(self.decrypt_title(&m.title))));
648
649        let modify_avatar = modify_avatar
650            .into_iter()
651            .map(|m| Ok(GroupChange::Avatar(m.avatar)));
652
653        let modify_description = modify_description.into_iter().map(|m| {
654            Ok(GroupChange::Description(
655                self.decrypt_description_text(&m.description),
656            ))
657        });
658
659        let modify_disappearing_message_timer =
660            modify_disappearing_message_timer.into_iter().map(|m| {
661                Ok(GroupChange::Timer(
662                    self.decrypt_disappearing_messages_timer(&m.timer),
663                ))
664            });
665
666        let modify_attributes_access =
667            modify_attributes_access.into_iter().map(|m| {
668                Ok(GroupChange::AttributeAccess(
669                    m.attributes_access.try_into()?,
670                ))
671            });
672
673        let modify_member_access = modify_member_access.into_iter().map(|m| {
674            Ok(GroupChange::MemberAccess(m.members_access.try_into()?))
675        });
676
677        let add_members_banned = add_members_banned
678            .into_iter()
679            .filter_map(|m| m.added)
680            .map(|m| {
681                Ok(GroupChange::AddBannedMember(self.decrypt_banned_member(m)?))
682            });
683
684        let delete_members_banned =
685            delete_members_banned.into_iter().map(|m| {
686                Ok(GroupChange::DeleteBannedMember(
687                    self.decrypt_service_id(&m.deleted_user_id)?,
688                ))
689            });
690
691        let promote_members_pending_pni_aci_profile_key =
692            promote_members_pending_pni_aci_profile_key
693                .into_iter()
694                .map(|m| {
695                    let promoted =
696                        self.decrypt_pni_aci_promotion_presentation(&m)?;
697                    Ok(GroupChange::PromotePendingPniAciMemberProfileKey(
698                        promoted,
699                    ))
700                });
701
702        let modify_add_from_invite_link_access =
703            modify_add_from_invite_link_access.into_iter().map(|m| {
704                Ok(GroupChange::InviteLinkAccess(
705                    m.add_from_invite_link_access.try_into()?,
706                ))
707            });
708
709        let add_members_pending_admin_approval =
710            add_members_pending_admin_approval
711                .into_iter()
712                .filter_map(|m| m.added)
713                .map(|added| {
714                    Ok(GroupChange::NewRequestingMember(
715                        self.decrypt_requesting_member(added)?,
716                    ))
717                });
718
719        let delete_members_pending_admin_approval =
720            delete_members_pending_admin_approval.into_iter().map(|m| {
721                Ok(GroupChange::DeleteRequestingMember(
722                    self.decrypt_aci(&m.deleted_user_id)?,
723                ))
724            });
725
726        let promote_members_pending_admin_approval =
727            promote_members_pending_admin_approval.into_iter().map(|m| {
728                Ok(GroupChange::PromoteRequestingMember {
729                    aci: self.decrypt_aci(&m.user_id)?,
730                    role: m.role.try_into()?,
731                })
732            });
733
734        let modify_invite_link_password =
735            modify_invite_link_password.into_iter().map(|m| {
736                Ok(GroupChange::InviteLinkPassword(
737                    BASE64_RELAXED.encode(m.invite_link_password),
738                ))
739            });
740
741        let modify_announcements_only = modify_announcements_only
742            .into_iter()
743            .map(|m| Ok(GroupChange::AnnouncementOnly(m.announcements_only)));
744
745        let modify_member_labels = modify_member_labels.into_iter().map(|m| {
746            Ok(GroupChange::MemberLabel {
747                user_id: self.decrypt_service_id(&m.user_id)?,
748                label_emoji: self.decrypt_member_label_emoji(&m.label_emoji),
749                label_string: self.decrypt_member_label_text(&m.label_string),
750            })
751        });
752
753        let modify_member_label_access =
754            modify_member_label_access.into_iter().map(|m| {
755                Ok(GroupChange::MemberLabelAccess(
756                    m.member_label_access.try_into()?,
757                ))
758            });
759
760        let terminate_group = terminate_group.into_iter().map(|_m| {
761            // The *presence* of the field signifies termination.
762            // TerminateGroupAction is a unit struct.
763            Ok(GroupChange::TerminateGroup)
764        });
765
766        let changes: Result<Vec<GroupChange>, GroupDecodingError> = new_members
767            .chain(delete_members)
768            .chain(modify_member_roles)
769            .chain(modify_member_profile_keys)
770            .chain(add_members_pending_profile_key)
771            .chain(delete_members_pending_profile_key)
772            .chain(promote_members_pending_profile_key)
773            .chain(modify_title)
774            .chain(modify_avatar)
775            .chain(modify_disappearing_message_timer)
776            .chain(modify_attributes_access)
777            .chain(modify_description)
778            .chain(modify_member_access)
779            .chain(add_members_banned)
780            .chain(delete_members_banned)
781            .chain(promote_members_pending_pni_aci_profile_key)
782            .chain(modify_add_from_invite_link_access)
783            .chain(add_members_pending_admin_approval)
784            .chain(delete_members_pending_admin_approval)
785            .chain(promote_members_pending_admin_approval)
786            .chain(modify_invite_link_password)
787            .chain(modify_announcements_only)
788            .chain(modify_member_labels)
789            .chain(modify_member_label_access)
790            .chain(terminate_group)
791            .collect();
792
793        Ok(GroupChanges {
794            group_id: group_id
795                .try_into()
796                .map_err(|_| GroupDecodingError::WrongBlob)?,
797            editor: source_user_id,
798            version,
799            changes: changes?,
800            change_epoch,
801        })
802    }

1048    /// Encrypt a group for creation, using credentials for member presentations.
1049    ///
1050    /// This method properly populates the `presentation` field for members with
1051    /// credentials, which is required by the Signal server for group creation.
1052    ///
1053    /// Members with credentials get added with presentations (full members).
1054    /// Members without credentials get added as pending invites.
1055    ///
1056    /// # Arguments
1057    /// * `title` - The group title
1058    /// * `description` - Optional group description
1059    /// * `disappearing_messages_timer` - Optional disappearing messages timer
1060    /// * `access_control` - Optional access control settings
1061    /// * `self_credential` - The creator's own credential (required)
1062    /// * `avatar_url` - The group avatar URL
1063    /// * `member_candidates` - Other members to add, with optional credentials
1064    /// * `server_public_params` - Server public params for creating presentations
1065    /// * `rng` - Random number generator
1066    #[allow(clippy::too_many_arguments)]
1067    pub fn encrypt_group_with_credentials<R: rand::Rng + rand::CryptoRng>(
1068        &self,
1069        title: &str,
1070        description: Option<&str>,
1071        disappearing_messages_timer: Option<&Timer>,
1072        access_control: Option<&AccessControl>,
1073        self_credential: &ExpiringProfileKeyCredential,
1074        member_candidates: &[GroupMemberCandidate],
1075        server_public_params: &ServerPublicParams,
1076        avatar_url: String,
1077        rng: &mut R,
1078    ) -> Result<proto::Group, GroupDecodingError> {
1079        let mut members = Vec::new();
1080        let mut members_pending_profile_key = Vec::new();
1081
1082        // Add self as administrator with presentation
1083        let self_presentation = self
1084            .create_member_presentation(server_public_params, self_credential);
1085        members.push(proto::Member {
1086            user_id: vec![],     // Server extracts from presentation
1087            profile_key: vec![], // Server extracts from presentation
1088            presentation: self_presentation,
1089            role: proto::member::Role::Administrator.into(),
1090            joined_at_version: 0,
1091            label_emoji: vec![],
1092            label_string: vec![],
1093        });
1094
1095        // Add other members
1096        for candidate in member_candidates {
1097            if let Some(credential) = &candidate.credential {
1098                // Has credential - add as full member with presentation
1099                let presentation = self.create_member_presentation(
1100                    server_public_params,
1101                    credential,
1102                );
1103                members.push(proto::Member {
1104                    user_id: vec![],
1105                    profile_key: vec![],
1106                    presentation,
1107                    role: proto::member::Role::Default.into(),
1108                    joined_at_version: 0,
1109                    label_emoji: vec![],
1110                    label_string: vec![],
1111                });
1112            } else {
1113                // No credential - add as pending invite
1114                let user_id_ciphertext =
1115                    self.encrypt_service_id(candidate.service_id)?;
1116                let self_aci = self_credential.aci();
1117                members_pending_profile_key.push(
1118                    proto::MemberPendingProfileKey {
1119                        member: Some(proto::Member {
1120                            user_id: user_id_ciphertext,
1121                            profile_key: vec![],
1122                            presentation: vec![],
1123                            role: proto::member::Role::Default.into(),
1124                            joined_at_version: 0,
1125                            label_emoji: vec![],
1126                            label_string: vec![],
1127                        }),
1128                        added_by_user_id: self.encrypt_aci(self_aci)?,
1129                        timestamp: 0, // Server sets
1130                    },
1131                );
1132            }
1133        }
1134
1135        // Encrypt title, description, timer
1136        let encrypted_title = self.encrypt_title(title, rng);
1137        let encrypted_description = self.encrypt_description(description, rng);
1138        let encrypted_timer = self.encrypt_disappearing_messages_timer(
1139            disappearing_messages_timer,
1140            rng,
1141        );
1142
1143        // Convert access control
1144        let proto_access_control =
1145            access_control.map(|ac| proto::AccessControl {
1146                attributes: ac.attributes.into(),
1147                members: ac.members.into(),
1148                add_from_invite_link: ac.add_from_invite_link.into(),
1149                member_label: ac.member_label.into(),
1150            });
1151
1152        Ok(proto::Group {
1153            public_key: zkgroup::serialize(
1154                &self.group_secret_params.get_public_params(),
1155            ),
1156            title: encrypted_title,
1157            avatar_url,
1158            disappearing_messages_timer: encrypted_timer,
1159            access_control: proto_access_control,
1160            version: 0,
1161            members,
1162            members_pending_profile_key,
1163            members_pending_admin_approval: vec![],
1164            invite_link_password: vec![],
1165            description: encrypted_description,
1166            announcements_only: false,
1167            members_banned: vec![],
1168            terminated: false,
1169        })
1170    }
