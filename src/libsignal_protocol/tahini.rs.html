6//! Tahini session management — pure state-machine policy.
7//!
8//! This module owns the Tahini protocol's policy decisions for the per-device
9//! [`SessionRecord`]: when to archive, when to break a deadlock, how to resolve
10//! tiebreakers on simultaneous initiation, when to mark a locally-initiated
11//! session as acknowledged, and how direction tracking flips between sends and
12//! receives.
13//!
14//! ## Boundary
15//!
16//! Functions here are **pure**: synchronous, no `async`, no store traits. They
17//! operate over an in-memory [`SessionRecord`] (or [`SessionState`]) and return
18//! decisions to the caller. Anything that talks to a `dyn *Store` lives in
19//! [`crate::session_management`].
20//!
21//! ## Verbs
22//!
23//! Callers invoke a verb that names the protocol event:
24//!
25//! - [`on_send`] — about to encrypt a message.
26//! - [`on_receive_active`] — successfully decrypted on the active session.
27//! - [`on_receive_inactive`] — successfully decrypted on an archived session
28//!   (also fires the deadlock breaker if the threshold is exceeded).
29//! - [`on_receive_session_state`] — per-state bookkeeping on any successful
30//!   decrypt: once the peer responds, clears the unacknowledged pre-key (so we
31//!   stop wrapping outgoing messages) and promotes `LocalNew` → `LocalAcked`.
32//! - [`on_archive`] — per-state bookkeeping when a session is retired into the
33//!   archive: clears the unacknowledged pre-key so its secret material does not
34//!   linger past the point we know it will not be used again.
35//! - [`on_initiate`] — locally-initiated session built from a peer's pre-key
36//!   bundle is about to be installed.
37//! - [`decide_initiation_placement`] / [`commit_received_initiation`] —
38//!   peer-initiated session built from an incoming PreKey message: first ask
39//!   where it should land (pure, read-only), then install it at that slot
40//!   (stamps origin `Remote`).

45// ── Policy constants ─────────────────────────────────────────────────────────
46
47/// Maximum number of Send ↔ ReceiveInactive direction changes before the
48/// deadlock detector archives the active session and forces re-establishment.
49pub(crate) const MAX_DIR_CHANGES: u32 = 5;
50
51// ── Types ────────────────────────────────────────────────────────────────────
52
53/// How a session was established, from the local device's perspective.
54///
55/// `Legacy` indicates a pre-Tahini session (proto default). Tahini policy keys
56/// "known by peer" off the pending pre-key rather than this field, so a Legacy
57/// session behaves like `Remote` once acknowledged, or like `LocalNew` while it
58/// still carries an unacknowledged pre-key.
59pub(crate) use crate::proto::storage::session_structure::Origin as SessionOrigin;
60/// The most recent action taken on a device record.
61///
62/// Used by Tahini's deadlock detector to track direction changes.
63pub(crate) use crate::proto::storage::tahini_session_management_state::LastAct;
64
65/// Where a newly-installed peer-initiated session landed in the device record.
66///
67/// Produced by [`decide_initiation_placement`] and consumed by
68/// [`commit_received_initiation`], so the caller can route the inner
69/// SignalMessage decryption against the correct session.
70#[derive(Debug, Clone, Copy, PartialEq, Eq)]
71pub(crate) enum NewSessionLocation {
72    Active,
73    Archived,
74}
75
76// ── Send path ────────────────────────────────────────────────────────────────
77
78/// Update direction-change state in advance of an encrypt.
79///
80/// If the previous action was a receive on an inactive session, this counts as
81/// a direction change. The actual deadlock-breaking archive happens on the
82/// receive path; send only updates the counter and the action marker.
83pub(crate) fn on_send(record: &mut SessionRecord) {
84    if record.last_act() == LastAct::ReceiveInactive {
85        record.increment_direction_change_ctr();
86    }
87    record.set_last_act(LastAct::Send);
88}
89
90// ── Receive path ─────────────────────────────────────────────────────────────
91
92/// Mark a successful decrypt on the active session.
93pub(crate) fn on_receive_active(record: &mut SessionRecord) {
94    record.set_last_act(LastAct::ReceiveActive);
95}
96
97/// Mark a successful decrypt on an archived (inactive) session.
98///
99/// If the prior action was a send, increments the direction-change counter.
100/// If the counter then exceeds [`MAX_DIR_CHANGES`], archives the active
101/// session and resets the counter (deadlock break).
102pub(crate) fn on_receive_inactive(
103    record: &mut SessionRecord,
104) -> Result<(), crate::SignalProtocolError> {
105    if record.last_act() == LastAct::Send {
106        record.increment_direction_change_ctr();
107    }
108    record.set_last_act(LastAct::ReceiveInactive);
109
110    if record.direction_change_ctr() > MAX_DIR_CHANGES {
111        log::info!(
112            "Tahini deadlock detected (direction_change_ctr={}), \
113             archiving active session on receive",
114            record.direction_change_ctr()
115        );
116        record.archive_current_state()?;
117        record.reset_direction_change_ctr();
118    }
119
120    Ok(())
121}
122
123/// Per-state bookkeeping after any successful decrypt, regardless of which
124/// session decrypted.
125///
126/// A successful decrypt means the peer has responded. If this session was not
127/// yet known to the peer (it still carries an unacknowledged pre-key), it is
128/// now acknowledged: drop the pre-key payload so we stop wrapping outgoing
129/// messages as pre-key messages, and record the transition in the origin for
130/// diagnostics. Complements the record-level [`on_receive_active`] /
131/// [`on_receive_inactive`] helpers.
132pub(crate) fn on_receive_session_state(state: &mut SessionState) {
133    if !state.is_known_by_peer() {
134        debug_assert!(
135            matches!(
136                state.origin(),
137                SessionOrigin::LocalNew | SessionOrigin::Legacy
138            ),
139            "is_known_by_peer is false but origin is {:?}",
140            state.origin()
141        );
142        state.clear_unacknowledged_pre_key_message();
143        if state.origin() == SessionOrigin::LocalNew {
144            state.set_origin(SessionOrigin::LocalAcked);
145        }
146    }
147}
148
149/// Per-state bookkeeping when a session is being archived.
150///
151/// Clears the unacknowledged pre-key payload. Archived sessions are never sent
152/// on and are never promoted back to active, so the pre-key material will not
153/// be used again — and because it holds cryptographic secrets, it must be
154/// deleted as soon as we know it can be deleted safely.
155pub(crate) fn on_archive(state: &mut SessionState) {
156    state.clear_unacknowledged_pre_key_message();
157}
158
159// ── Session establishment ────────────────────────────────────────────────────
160
161/// Install a locally-initiated session as the active session.
162///
163/// Stamps origin `LocalNew`, archives any prior active session, installs
164/// `new_session` as active, and clears the deadlock-detector history for the
165/// fresh session: the direction-change counter goes to zero and `last_act` is
166/// cleared to [`LastAct::Unset`].
167pub(crate) fn on_initiate(record: &mut SessionRecord, mut new_session: SessionState) {
168    new_session.set_origin(SessionOrigin::LocalNew);
169    record.replace_active_session(new_session);
170    record.reset_direction_change_ctr();
171    record.set_last_act(LastAct::Unset);
172}
173
174/// Decide where a peer-initiated session built from an incoming PreKey
175/// message should land in the device record.
176///
177/// Pure: reads `record` and returns a placement decision. Does not mutate.
178///
179/// - If there is no active session, the active session is already known by
180///   the peer, or the incoming identity differs from the active session's peer
181///   identity, the new session goes [`Active`](NewSessionLocation::Active).
182///   Identity replacement takes precedence over the SID tiebreaker: a session
183///   targeting the retired identity may no longer be usable by the peer.
184/// - If the active session is not yet known by the peer (an unacknowledged
185///   locally-initiated session — `LocalNew`, or a migrated pre-Tahini session
186///   still carrying a pending pre-key), and the identities match, this is a
187///   simultaneous initiation: compare session IDs lexicographically. The higher
188///   SID wins the active slot; the loser is destined for the archive.
189///
190/// Deduplication is handled separately by the caller (a session with this ID
191/// may already exist, in which case the caller skips placement entirely).
192pub(crate) fn decide_initiation_placement(
193    record: &SessionRecord,
194    incoming_session_id: &[u8],
195    incoming_identity: &IdentityKey,
196) -> Result<NewSessionLocation, InvalidSessionError> {
197    Ok(match record.active_session() {
198        None => NewSessionLocation::Active,
199        Some(active) => {
200            // The caller handles deduplication before reaching here, so the incoming
201            // session ID must differ from the active session's. Session IDs are
202            // public, so a plain (non-constant-time) comparison is fine.
203            debug_assert_ne!(
204                incoming_session_id,
205                active.session_id(),
206                "incoming session ID matches active session; dedup should have caught this"
207            );
208            if active.is_known_by_peer()
209                || active.remote_identity_key()? != Some(*incoming_identity)
210                || incoming_session_id > active.session_id()
211            {
212                NewSessionLocation::Active
213            } else {
214                NewSessionLocation::Archived
215            }
216        }
217    })
218}
219
220/// Install a peer-initiated session at the previously-decided slot, after
221/// AEAD has verified.
222///
223/// Stamps origin `Remote`, writes `new_session` into the slot indicated by
224/// `placement`, and applies post-decrypt bookkeeping:
225/// - On `Active`: the active session is replaced, so its deadlock history is
226///   reset — direction-change counter to zero and `last_act = ReceiveActive`
227///   (the winning initiation was received on the now-active session).
228/// - On `Archived` (tiebreaker loser): the active session is unchanged, so the
229///   deadlock history is left untouched by placement. The caller must then
230///   invoke [`on_receive_inactive`] for the successfully decrypted enclosed
231///   message; that receive can increment the counter and fire the breaker.
232pub(crate) fn commit_received_initiation(
233    record: &mut SessionRecord,
234    mut new_session: SessionState,
235    placement: NewSessionLocation,
236) -> Result<(), crate::SignalProtocolError> {
237    new_session.set_origin(SessionOrigin::Remote);
238    match placement {
239        NewSessionLocation::Active => {
240            record.replace_active_session(new_session);
241            record.reset_direction_change_ctr();
242            on_receive_active(record);
243            Ok(())
244        }
245        NewSessionLocation::Archived => {
246            record.archive_session(new_session);
247            Ok(())
248        }
249    }
250}

340    #[test]
341    fn commit_archived_leaves_deadlock_state_untouched() {
342        // Placement alone must not advance the counter or fire the breaker.
343        // The caller subsequently records the decrypted enclosed message as
344        // an inactive receive; full receive tests cover that separate step.
345        let mut record =
346            SessionRecord::new(SessionState::from_session_structure(SessionStructure {
347                alice_base_key: vec![0x05, 0xff],
348                ..Default::default()
349            }));
350        record.set_last_act(LastAct::Send);
351        for _ in 0..MAX_DIR_CHANGES {
352            record.increment_direction_change_ctr();
353        }
354        assert_eq!(record.direction_change_ctr(), MAX_DIR_CHANGES);
355        let active_before = record.active_session().map(|s| s.session_id().to_vec());
356
357        // The loser: a lower session id that lost the tiebreak in the caller.
358        let loser = SessionState::from_session_structure(SessionStructure {
359            alice_base_key: vec![0x05, 0x01],
360            ..Default::default()
361        });
362        commit_received_initiation(&mut record, loser, NewSessionLocation::Archived).unwrap();
363
364        // Counter and marker unchanged; the active session was NOT archived by a
365        // spurious break; only the loser sits in the archive.
366        assert_eq!(record.direction_change_ctr(), MAX_DIR_CHANGES);
367        assert_eq!(record.last_act(), LastAct::Send);
368        assert_eq!(
369            record.active_session().map(|s| s.session_id().to_vec()),
370            active_before
371        );
372        assert_eq!(record.archived_sessions().count(), 1);
373    }

438    #[test]
439    fn cross_upgrade_simultaneous_init_converges() {
440        use crate::proto::storage::session_structure::PendingPreKey;
441
442        // Both sides initiate, and their pre-key messages cross. One side is a
443        // full Tahini session (origin LocalNew); the other is a migrated
444        // pre-Tahini session (origin reads as Legacy) that still carries a
445        // pending pre-key. Both must run the SID tiebreak and converge on the
446        // same active session (the higher base key wins on both sides). Before
447        // is_known_by_peer keyed on the pending pre-key, the Legacy side yielded
448        // unconditionally and the two could diverge.
449
450        // Returns the SID of whichever session ends up active after receiving a
451        // crossing initiation: the incoming one if it takes the active slot,
452        // otherwise our own.
453        let active_after = |origin: SessionOrigin, own_sid: &[u8], incoming: &[u8]| -> Vec<u8> {
454            let record =
455                SessionRecord::new(SessionState::from_session_structure(SessionStructure {
456                    origin: origin.into(),
457                    alice_base_key: own_sid.to_vec(),
458                    remote_identity_public: vec![0x05; 33],
459                    pending_pre_key: Some(PendingPreKey::default()),
460                    ..Default::default()
461                }));
462            match decide_initiation_placement(
463                &record,
464                incoming,
465                &IdentityKey::decode(&[0x05; 33]).unwrap(),
466            )
467            .unwrap()
468            {
469                NewSessionLocation::Active => incoming.to_vec(),
470                NewSessionLocation::Archived => own_sid.to_vec(),
471            }
472        };
473
474        for (alice_sid, bob_sid) in [
475            (vec![0x01], vec![0x02]), // Bob's SID higher
476            (vec![0x02], vec![0x01]), // Alice's SID higher
477        ] {
478            let alice_active = active_after(SessionOrigin::Legacy, &alice_sid, &bob_sid);
479            let bob_active = active_after(SessionOrigin::LocalNew, &bob_sid, &alice_sid);
480            // Convergence: both sides pick the same active session...
481            assert_eq!(alice_active, bob_active, "sids {alice_sid:?}/{bob_sid:?}");
482            // ...and it is the higher SID.
483            assert_eq!(alice_active, std::cmp::max(alice_sid, bob_sid));
484        }
485    }
