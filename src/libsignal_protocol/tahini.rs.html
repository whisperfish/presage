6//! Tahini session management — pure state-machine policy.
7//!
8//! This module owns the Tahini protocol's policy decisions for the per-device
9//! [`SessionRecord`]: when to archive, when to break a deadlock, how to resolve
10//! tiebreakers on simultaneous initiation, when to mark a locally-initiated
11//! session as acknowledged, and how direction tracking flips between sends and
12//! receives.
13//!
14//! ## Boundary
15//!
16//! Functions here are **pure**: synchronous, no `async`, no store traits. They
17//! operate over an in-memory [`SessionRecord`] (or [`SessionState`]) and return
18//! decisions to the caller. Anything that talks to a `dyn *Store` lives in
19//! [`crate::session_management`].
20//!
21//! ## Verbs
22//!
23//! Callers invoke a verb that names the protocol event:
24//!
25//! - [`on_send`] — about to encrypt a message.
26//! - [`on_receive_active`] — successfully decrypted on the active session.
27//! - [`on_receive_inactive`] — successfully decrypted on an archived session
28//!   (also fires the deadlock breaker if the threshold is exceeded).
29//! - [`on_receive_session_state`] — per-state bookkeeping on any successful
30//!   decrypt: once the peer responds, clears the unacknowledged pre-key (so we
31//!   stop wrapping outgoing messages) and promotes `LocalNew` → `LocalAcked`.
32//! - [`may_match_retransmitted_initiation`] — whether an incoming PreKey
33//!   message may be matched to an existing session as a retransmission.
34//! - [`on_archive`] — per-state bookkeeping when a session is retired into the
35//!   archive: clears the unacknowledged pre-key so its secret material does not
36//!   linger past the point we know it will not be used again.
37//! - [`on_initiate`] — locally-initiated session built from a peer's pre-key
38//!   bundle is about to be installed.
39//! - [`decide_initiation_placement`] / [`commit_received_initiation`] —
40//!   peer-initiated session built from an incoming PreKey message: first ask
41//!   where it should land (pure, read-only), then install it at that slot
42//!   (stamps origin `Remote`).
43

47// ── Policy constants ─────────────────────────────────────────────────────────
48
49/// Maximum number of Send ↔ ReceiveInactive direction changes before the
50/// deadlock detector archives the active session and forces re-establishment.
51pub(crate) const MAX_DIR_CHANGES: u32 = 5;
52
53// ── Types ────────────────────────────────────────────────────────────────────
54
55/// How a session was established, from the local device's perspective.
56///
57/// `Legacy` indicates a pre-Tahini session (proto default). Tahini policy keys
58/// "known by peer" off the pending pre-key rather than this field, so a Legacy
59/// session behaves like `Remote` once acknowledged, or like `LocalNew` while it
60/// still carries an unacknowledged pre-key.
61pub(crate) use crate::proto::storage::session_structure::Origin as SessionOrigin;
62/// The most recent action taken on a device record.
63///
64/// Used by Tahini's deadlock detector to track direction changes.
65pub(crate) use crate::proto::storage::tahini_session_management_state::LastAct;
66
67/// Where a newly-installed peer-initiated session landed in the device record.
68///
69/// Produced by [`decide_initiation_placement`] and consumed by
70/// [`commit_received_initiation`], so the caller can route the inner
71/// SignalMessage decryption against the correct session.
72#[derive(Debug, Clone, Copy, PartialEq, Eq)]
73pub(crate) enum NewSessionLocation {
74    Active,
75    Archived,
76}
77
78// ── Send path ────────────────────────────────────────────────────────────────
79
80/// Update direction-change state in advance of an encrypt.
81///
82/// If the previous action was a receive on an inactive session, this counts as
83/// a direction change. The actual deadlock-breaking archive happens on the
84/// receive path; send only updates the counter and the action marker.
85pub(crate) fn on_send(record: &mut SessionRecord) {
86    if record.last_act() == LastAct::ReceiveInactive {
87        record.increment_direction_change_ctr();
88    }
89    record.set_last_act(LastAct::Send);
90}
91
92// ── Receive path ─────────────────────────────────────────────────────────────
93
94/// Mark a successful decrypt on the active session.
95pub(crate) fn on_receive_active(record: &mut SessionRecord) {
96    record.set_last_act(LastAct::ReceiveActive);
97}
98
99/// Mark a successful decrypt on an archived (inactive) session.
100///
101/// If the prior action was a send, increments the direction-change counter.
102/// If the counter then exceeds [`MAX_DIR_CHANGES`], archives the active
103/// session and resets the counter (deadlock break).
104pub(crate) fn on_receive_inactive(
105    record: &mut SessionRecord,
106) -> Result<(), crate::SignalProtocolError> {
107    if record.last_act() == LastAct::Send {
108        record.increment_direction_change_ctr();
109    }
110    record.set_last_act(LastAct::ReceiveInactive);
111
112    if record.direction_change_ctr() > MAX_DIR_CHANGES {
113        log::info!(
114            "Tahini deadlock detected (direction_change_ctr={}), \
115             archiving active session on receive",
116            record.direction_change_ctr()
117        );
118        record.archive_current_state()?;
119        record.reset_direction_change_ctr();
120    }
121
122    Ok(())
123}
124
125/// Per-state bookkeeping after any successful decrypt, regardless of which
126/// session decrypted.
127///
128/// A successful decrypt means the peer has responded. If this session was not
129/// yet known to the peer (it still carries an unacknowledged pre-key), it is
130/// now acknowledged: drop the pre-key payload so we stop wrapping outgoing
131/// messages as pre-key messages, and record the transition in the origin for
132/// diagnostics. Complements the record-level [`on_receive_active`] /
133/// [`on_receive_inactive`] helpers.
134pub(crate) fn on_receive_session_state(state: &mut SessionState) {
135    if !state.is_known_by_peer() {
136        debug_assert!(
137            matches!(
138                state.origin(),
139                SessionOrigin::LocalNew | SessionOrigin::Legacy
140            ),
141            "is_known_by_peer is false but origin is {:?}",
142            state.origin()
143        );
144        state.clear_unacknowledged_pre_key_message();
145        if state.origin() == SessionOrigin::LocalNew {
146            state.set_origin(SessionOrigin::LocalAcked);
147        }
148    }
149}
150
151/// Whether an incoming PreKey message whose base key matches `state` may be
152/// treated as a retransmitted initiation for it.
153pub(crate) fn may_match_retransmitted_initiation(state: &SessionState) -> bool {
154    match state.origin() {
155        SessionOrigin::LocalNew | SessionOrigin::LocalAcked => false,
156        SessionOrigin::Remote => true,
157        SessionOrigin::Legacy => state.is_known_by_peer(),
158    }
159}
160
161/// Per-state bookkeeping when a session is being archived.
162///
163/// Clears the unacknowledged pre-key payload. Archived sessions are never sent
164/// on and are never promoted back to active, so the pre-key material will not
165/// be used again — and because it holds cryptographic secrets, it must be
166/// deleted as soon as we know it can be deleted safely.
167pub(crate) fn on_archive(state: &mut SessionState) {
168    state.clear_unacknowledged_pre_key_message();
169}
170
171// ── Session establishment ────────────────────────────────────────────────────
172
173/// Install a locally-initiated session as the active session.
174///
175/// Stamps origin `LocalNew`, archives any prior active session, installs
176/// `new_session` as active, and clears the deadlock-detector history for the
177/// fresh session: the direction-change counter goes to zero and `last_act` is
178/// cleared to [`LastAct::Unset`].
179pub(crate) fn on_initiate(record: &mut SessionRecord, mut new_session: SessionState) {
180    new_session.set_origin(SessionOrigin::LocalNew);
181    record.replace_active_session(new_session);
182    record.reset_direction_change_ctr();
183    record.set_last_act(LastAct::Unset);
184}
185
186/// Decide where a peer-initiated session built from an incoming PreKey
187/// message should land in the device record.
188///
189/// Pure: reads `record` and returns a placement decision. Does not mutate.
190///
191/// - If there is no active session, the active session is already known by
192///   the peer, or the incoming identity differs from the active session's peer
193///   identity, the new session goes [`Active`](NewSessionLocation::Active).
194///   Identity replacement takes precedence over the SID tiebreaker: a session
195///   targeting the retired identity may no longer be usable by the peer.
196/// - If the active session is not yet known by the peer (an unacknowledged
197///   locally-initiated session — `LocalNew`, or a migrated pre-Tahini session
198///   still carrying a pending pre-key), and the identities match, this is a
199///   simultaneous initiation: compare session IDs lexicographically. The higher
200///   SID wins the active slot; the loser is destined for the archive.
201///
202/// Deduplication is handled separately by the caller (a session with this ID
203/// may already exist, in which case the caller skips placement entirely).
204pub(crate) fn decide_initiation_placement(
205    record: &SessionRecord,
206    incoming_session_id: &[u8],
207    incoming_identity: &IdentityKey,
208) -> Result<NewSessionLocation, InvalidSessionError> {
209    Ok(match record.active_session() {
210        None => NewSessionLocation::Active,
211        Some(active) => {
212            // The caller handles deduplication before reaching here, so the incoming
213            // session ID must differ from the active session's. Session IDs are
214            // public, so a plain (non-constant-time) comparison is fine.
215            debug_assert_ne!(
216                incoming_session_id,
217                active.session_id(),
218                "incoming session ID matches active session; dedup should have caught this"
219            );
220            if active.is_known_by_peer()
221                || active.remote_identity_key()? != Some(*incoming_identity)
222                || incoming_session_id > active.session_id()
223            {
224                NewSessionLocation::Active
225            } else {
226                NewSessionLocation::Archived
227            }
228        }
229    })
230}
231
232/// Install a peer-initiated session at the previously-decided slot, after
233/// AEAD has verified.
234///
235/// Stamps origin `Remote`, writes `new_session` into the slot indicated by
236/// `placement`, and applies post-decrypt bookkeeping:
237/// - On `Active`: the active session is replaced, so its deadlock history is
238///   reset — direction-change counter to zero and `last_act = ReceiveActive`
239///   (the winning initiation was received on the now-active session).
240/// - On `Archived` (tiebreaker loser): the active session is unchanged, so the
241///   deadlock history is left untouched by placement. The caller must then
242///   invoke [`on_receive_inactive`] for the successfully decrypted enclosed
243///   message; that receive can increment the counter and fire the breaker.
244pub(crate) fn commit_received_initiation(
245    record: &mut SessionRecord,
246    mut new_session: SessionState,
247    placement: NewSessionLocation,
248) -> Result<(), crate::SignalProtocolError> {
249    new_session.set_origin(SessionOrigin::Remote);
250    match placement {
251        NewSessionLocation::Active => {
252            record.replace_active_session(new_session);
253            record.reset_direction_change_ctr();
254            on_receive_active(record);
255            Ok(())
256        }
257        NewSessionLocation::Archived => {
258            record.archive_session(new_session);
259            Ok(())
260        }
261    }
262}

352    #[test]
353    fn commit_archived_leaves_deadlock_state_untouched() {
354        // Placement alone must not advance the counter or fire the breaker.
355        // The caller subsequently records the decrypted enclosed message as
356        // an inactive receive; full receive tests cover that separate step.
357        let mut record =
358            SessionRecord::new(SessionState::from_session_structure(SessionStructure {
359                alice_base_key: vec![0x05, 0xff],
360                ..Default::default()
361            }));
362        record.set_last_act(LastAct::Send);
363        for _ in 0..MAX_DIR_CHANGES {
364            record.increment_direction_change_ctr();
365        }
366        assert_eq!(record.direction_change_ctr(), MAX_DIR_CHANGES);
367        let active_before = record.active_session().map(|s| s.session_id().to_vec());
368
369        // The loser: a lower session id that lost the tiebreak in the caller.
370        let loser = SessionState::from_session_structure(SessionStructure {
371            alice_base_key: vec![0x05, 0x01],
372            ..Default::default()
373        });
374        commit_received_initiation(&mut record, loser, NewSessionLocation::Archived).unwrap();
375
376        // Counter and marker unchanged; the active session was NOT archived by a
377        // spurious break; only the loser sits in the archive.
378        assert_eq!(record.direction_change_ctr(), MAX_DIR_CHANGES);
379        assert_eq!(record.last_act(), LastAct::Send);
380        assert_eq!(
381            record.active_session().map(|s| s.session_id().to_vec()),
382            active_before
383        );
384        assert_eq!(record.archived_sessions().count(), 1);
385    }

472    #[test]
473    fn cross_upgrade_simultaneous_init_converges() {
474        use crate::proto::storage::session_structure::PendingPreKey;
475
476        // Both sides initiate, and their pre-key messages cross. One side is a
477        // full Tahini session (origin LocalNew); the other is a migrated
478        // pre-Tahini session (origin reads as Legacy) that still carries a
479        // pending pre-key. Both must run the SID tiebreak and converge on the
480        // same active session (the higher base key wins on both sides). Before
481        // is_known_by_peer keyed on the pending pre-key, the Legacy side yielded
482        // unconditionally and the two could diverge.
483
484        // Returns the SID of whichever session ends up active after receiving a
485        // crossing initiation: the incoming one if it takes the active slot,
486        // otherwise our own.
487        let active_after = |origin: SessionOrigin, own_sid: &[u8], incoming: &[u8]| -> Vec<u8> {
488            let record =
489                SessionRecord::new(SessionState::from_session_structure(SessionStructure {
490                    origin: origin.into(),
491                    alice_base_key: own_sid.to_vec(),
492                    remote_identity_public: vec![0x05; 33],
493                    pending_pre_key: Some(PendingPreKey::default()),
494                    ..Default::default()
495                }));
496            match decide_initiation_placement(
497                &record,
498                incoming,
499                &IdentityKey::decode(&[0x05; 33]).unwrap(),
500            )
501            .unwrap()
502            {
503                NewSessionLocation::Active => incoming.to_vec(),
504                NewSessionLocation::Archived => own_sid.to_vec(),
505            }
506        };
507
508        for (alice_sid, bob_sid) in [
509            (vec![0x01], vec![0x02]), // Bob's SID higher
510            (vec![0x02], vec![0x01]), // Alice's SID higher
511        ] {
512            let alice_active = active_after(SessionOrigin::Legacy, &alice_sid, &bob_sid);
513            let bob_active = active_after(SessionOrigin::LocalNew, &bob_sid, &alice_sid);
514            // Convergence: both sides pick the same active session...
515            assert_eq!(alice_active, bob_active, "sids {alice_sid:?}/{bob_sid:?}");
516            // ...and it is the higher SID.
517            assert_eq!(alice_active, std::cmp::max(alice_sid, bob_sid));
518        }
519    }
