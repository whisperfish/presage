12#[repr(u8)]
13#[derive(Copy, Clone, Debug, Eq, PartialEq, PartialDefault, derive_more::TryFrom)]
14#[try_from(repr)]
15pub enum GenericServerParamsVersion {
16    #[partial_default]
17    Legacy = 0,
18    Standard = 1,
19}
20
21#[derive(Clone, Serialize, Deserialize, PartialDefault)]
22pub struct GenericServerSecretParamsLegacy {
23    version: VersionByte<{ GenericServerParamsVersion::Legacy as u8 }>,
24    pub(crate) credential_key:
25        zkcredential::credentials::CredentialKeyPair<zkcredential::credentials::LegacyMode>,
26}
27
28impl GenericServerSecretParamsLegacy {
29    pub fn generate(randomness: RandomnessBytes) -> Self {
30        Self {
31            version: Default::default(),
32            credential_key: zkcredential::credentials::CredentialKeyPair::generate(randomness),
33        }
34    }
35
36    pub fn get_public_params(
37        &self,
38    ) -> GenericServerPublicParamsImpl<{ GenericServerParamsVersion::Legacy as u8 }> {
39        GenericServerPublicParamsImpl {
40            version: self.version,
41            credential_key: self.credential_key.public_key().clone(),
42        }
43    }
44}

46#[derive(Clone, Serialize, Deserialize, PartialDefault)]

71#[derive(Clone, Serialize, PartialDefault, derive_more::From)]
72#[serde(untagged)]
73pub enum GenericServerSecretParams {
74    #[partial_default]
75    V0(GenericServerSecretParamsLegacy),
76    V1(GenericServerSecretParamsStandard),
77}
78
79impl GenericServerSecretParams {
80    pub fn get_public_params(&self) -> GenericServerPublicParams {
81        match self {
82            GenericServerSecretParams::V0(params) => params.get_public_params().into(),
83            GenericServerSecretParams::V1(params) => params.get_public_params().into(),
84        }
85    }
86}
87
88impl TryFrom<&'_ [u8]> for GenericServerSecretParams {
89    type Error = ZkGroupDeserializationFailure;
90    fn try_from(bytes: &[u8]) -> Result<Self, ZkGroupDeserializationFailure> {
91        match bytes
92            .first()
93            .and_then(|&v| v.try_into().ok())
94            .ok_or(ZkGroupDeserializationFailure::new::<Self>())?
95        {
96            GenericServerParamsVersion::Legacy => {
97                crate::deserialize::<GenericServerSecretParamsLegacy>(bytes).map(Self::V0)
98            }
99            GenericServerParamsVersion::Standard => {
100                crate::deserialize::<GenericServerSecretParamsStandard>(bytes).map(Self::V1)
101            }
102        }
103    }
104}

106// zkcredential public keys don't have different modes, but it's still useful to track how they were
107// generated so that we can early-exit if a client uses public params that definitely won't match a
108// given credential.
109#[derive(Clone, Serialize, Deserialize, PartialDefault)]
110pub struct GenericServerPublicParamsImpl<const V: u8> {
111    version: VersionByte<V>,
112    pub(crate) credential_key: zkcredential::credentials::CredentialPublicKey,
113}
114
115#[derive(Clone, Serialize, PartialDefault, derive_more::From)]
116#[serde(untagged)]
117pub enum GenericServerPublicParams {
118    #[partial_default]
119    V0(GenericServerPublicParamsImpl<{ GenericServerParamsVersion::Legacy as u8 }>),
120    V1(GenericServerPublicParamsImpl<{ GenericServerParamsVersion::Standard as u8 }>),
121}
