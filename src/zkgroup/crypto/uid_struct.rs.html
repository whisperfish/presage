17#[derive(Copy, Clone, PartialEq, Eq, Serialize, Deserialize, PartialDefault)]
18pub struct UidStruct {
19    // Currently unused. It would be possible to convert this back to the correct kind of ServiceId
20    // using the same technique as decryption: comparing possible M1 points and seeing which one
21    // matches. But we don't have a need for that, and therefore it's better if that operation
22    // remains part of decryption, so that you're guaranteed to get a valid result or an error in
23    // one step.
24    //
25    // At the same time, we can't just remove the field: it's serialized as part of AuthCredential
26    // and AuthCredentialWithPni, which clients store locally.
27    #[serde(rename = "bytes")]
28    raw_uuid_bytes: UidBytes,
29    pub(crate) M1: RistrettoPoint,
30    pub(crate) M2: RistrettoPoint,
31}
32
33impl UidStruct {
34    pub fn from_service_id(service_id: ServiceId) -> Self {
35        let M1 = Self::calc_M1(Self::seed_M1(), service_id);
36        let raw_uuid_bytes = service_id.raw_uuid().into_bytes();
37        let M2 = RistrettoPoint::lizard_encode::<Sha256>(&raw_uuid_bytes);
38        UidStruct {
39            raw_uuid_bytes,
40            M1,
41            M2,
42        }
43    }
44
45    pub(crate) fn for_false_pni_derived_from_aci(aci: libsignal_core::Aci, salt: &[u8]) -> Self {
46        // A v7 UUID generated on 2026-07-28.
47        const FALSE_PNI_NAMESPACE: uuid::Uuid = uuid::uuid!("019faa2b-7bcb-70c2-ac7f-cd770414689a");
48        // Hash the salt so that it less directly affects the older SHA-1 used by UUIDv5.
49        let hashed_salt = Sha256::digest(salt);
50        let false_pni = libsignal_core::Pni::from(new_v5_uuid(
51            &FALSE_PNI_NAMESPACE,
52            &hashed_salt,
53            uuid::Uuid::from(aci).as_bytes(),
54        ));
55        Self {
56            // Nothing reads this field; it's present only for debugging and backwards
57            // compatibility. Filling it with zeros marks that it's not useful anyway.
58            raw_uuid_bytes: uuid::Uuid::nil().into_bytes(),
59            ..Self::from_service_id(false_pni.into())
60        }
61    }
62
63    pub(crate) fn seed_M1() -> Sho {
64        Sho::new_seed(b"Signal_ZKGroup_20200424_UID_CalcM1")
65    }
66
67    pub(crate) fn calc_M1(mut seed: Sho, service_id: ServiceId) -> RistrettoPoint {
68        seed.absorb_and_ratchet(&service_id.service_id_binary());
69        seed.get_point()
70    }
71}
72
73impl zkcredential::attributes::Attribute for UidStruct {
74    fn as_points(&self) -> [RistrettoPoint; 2] {
75        [self.M1, self.M2]
76    }
77}

79/// A reimplementation of `uuid::Uuid::new_v5` using the RustCrypto `sha1` crate.
80///
81/// `first_name_part` and `second_name_part` are directly concatenated; as such, one or the other of
82/// them should be fixed-width to avoid "foo || bar" and "foob || ar" resulting in the same UUID.
83///
84/// See <https://www.rfc-editor.org/rfc/rfc9562.html#name-uuid-version-5>.
85fn new_v5_uuid(
86    namespace: &uuid::Uuid,
87    first_name_part: &[u8],
88    second_name_part: &[u8],
89) -> uuid::Uuid {
90    use sha1::Digest as _;
91    let uuid_v5_style_hash = sha1::Sha1::new()
92        .chain_update(namespace.as_bytes())
93        .chain_update(first_name_part)
94        .chain_update(second_name_part)
95        .finalize();
96    uuid::Builder::from_sha1_bytes(
97        *uuid_v5_style_hash
98            .first_chunk()
99            .expect("SHA-1 hashes are longer than UUIDs"),
100    )
101    .into_uuid()
102}

104#[cfg(test)]
105mod test {
106    use super::*;
107
108    #[test]
109    fn v5_uuid_test_vector_from_rfc() {
110        let dns_namespace = uuid::uuid!("6ba7b810-9dad-11d1-80b4-00c04fd430c8");
111        assert_eq!(
112            new_v5_uuid(&dns_namespace, b"www.example.com", b""),
113            uuid::uuid!("2ed6657d-e927-568b-95e1-2665a8aea6a2")
114        );
115        assert_eq!(
116            new_v5_uuid(&dns_namespace, b"www.", b"example.com"),
117            uuid::uuid!("2ed6657d-e927-568b-95e1-2665a8aea6a2")
118        );
119    }
