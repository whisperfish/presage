71impl ProfileKeyEncryptionDomain {
72    pub(crate) fn decrypt(
73        key_pair: &KeyPair,
74        ciphertext: &Ciphertext,
75        uid_bytes: UidBytes,
76    ) -> Result<ProfileKeyBytes, ZkGroupVerificationFailure> {
77        let M4 = key_pair
78            .decrypt_to_second_point(ciphertext)
79            .map_err(|_| ZkGroupVerificationFailure)?;
80        let candidates = M4.map_to_curve_inverse();
81
82        let target_M3 = key_pair.a1.invert() * ciphertext.as_points()[0];
83        Self::decrypt_from_candidates(
84            uid_bytes,
85            target_M3,
86            candidates
87                .first_chunk::<8>()
88                .expect("we only need the positive candidates"),
89        )
90    }
91
92    // Factored out for additional testing.
93    fn decrypt_from_candidates(
94        uid_bytes: UidBytes,
95        target_M3: RistrettoPoint,
96        candidates: &[CtOption<[u8; 32]>; 8],
97    ) -> Result<ProfileKeyBytes, ZkGroupVerificationFailure> {
98        let seed_sho = profile_key_struct::ProfileKeyStruct::seed_M3();
99
100        let mut retval = CtOption::new(ProfileKeyBytes::partial_default(), Choice::from(0u8));
101        // The number of valid solutions found. Note we would see if n_found > 1 because the closures in
102        // CtOption::and_then and CtOption::or_else always run.
103        let mut n_found = 0;
104        #[allow(clippy::needless_range_loop)]
105        for i in 0..candidates.len() {
106            retval = retval.or_else(|| {
107                candidates[i].and_then(|profile_key_bytes| {
108                    let mut candidate_retval =
109                        CtOption::new(ProfileKeyBytes::partial_default(), Choice::from(0u8));
110                    for j in 0..8 {
111                        candidate_retval = candidate_retval.or_else(|| {
112                            let mut pk = profile_key_bytes;
113                            if ((j >> 2) & 1) == 1 {
114                                pk[0] |= 0x01;
115                            }
116                            if ((j >> 1) & 1) == 1 {
117                                pk[31] |= 0x80;
118                            }
119                            if (j & 1) == 1 {
120                                pk[31] |= 0x40;
121                            }
122                            let M3 = profile_key_struct::ProfileKeyStruct::calc_M3(
123                                seed_sho.clone(),
124                                pk,
125                                uid_bytes,
126                            );
127                            let found = M3.ct_eq(&target_M3);
128                            n_found += (found & candidates[i].is_some()).unwrap_u8();
129                            CtOption::new(pk, found)
130                        });
131                    }
132                    candidate_retval
133                })
134            });
135        }
136        if n_found == 1 {
137            // We can unwrap because n_found > 0 implies that candidate_retval is Some, which means
138            // retval is Some
139            Ok(retval.unwrap())
140        } else {
141            Err(ZkGroupVerificationFailure)
142        }
143    }
144}

246    /// Not constant-time!
247    fn convert_option_to_ct_option<T: PartialDefault>(input: Option<T>) -> CtOption<T> {
248        match input {
249            Some(value) => CtOption::new(value, Choice::from(1)),
250            None => CtOption::new(T::partial_default(), Choice::from(0)),
251        }
252    }

308    fn test_edge_cases_with_all_zero_profile_keys(
309        candidates: [Option<[u8; 32]>; 8],
310    ) -> Result<ProfileKeyBytes, ZkGroupVerificationFailure> {
311        let uid_bytes = TEST_ARRAY_16;
312        // These cases rely on the behavior of `CtOption::and_then` exposing an all-zeros
313        // (`Default`) value when the option is in the `None` state. Therefore, we need a profile
314        // key that could conceivably be encoded as all-zeros, the simplest of which is...all-zeros.
315        let target_m3 = profile_key_struct::ProfileKeyStruct::calc_M3(
316            profile_key_struct::ProfileKeyStruct::seed_M3(),
317            ALL_ZEROS_PROFILE_KEY,
318            uid_bytes,
319        );
320        ProfileKeyEncryptionDomain::decrypt_from_candidates(
321            uid_bytes,
322            target_m3,
323            &candidates.map(convert_option_to_ct_option),
324        )
325    }
